import { Pool } from "pg";
import dotenv from "dotenv";
import path from "node:path";

dotenv.config({
  path: path.resolve(process.cwd(), "../.env"),
});

export const pool = new Pool({
  host: process.env.POSTGRES_HOST || "localhost",
  port: Number(process.env.POSTGRES_PORT) || 5433,
  database: process.env.POSTGRES_DB || "postgres",
  user: process.env.POSTGRES_USER || "lms_ai_user",
  password: process.env.POSTGRES_PASSWORD,
});

export const checkDatabaseConnection = async (): Promise<void> => {
  const client = await pool.connect();

  try {
    const result = await client.query("SELECT current_database() AS database");

    console.log("PostgreSQL connected:", result.rows[0].database);
  } finally {
    client.release();
  }
};

  host: process.env.REDIS_HOST || "localhost",
  port: Number(process.env.REDIS_PORT) || 6379,
  lazyConnect: true,
  maxRetriesPerRequest: 1,
});

export const checkRedisConnection = async (): Promise<void> => {
  await redis.connect();

  const result = await redis.ping();

  if (result !== "PONG") {
    throw new Error("Redis health check failed");
  }

  console.log("Redis connected:", result);
};

--user table----------------
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(180) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    avatar_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);
-----user roles table-------------
CREATE TABLE roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(30) UNIQUE NOT NULL
);

CREATE TABLE user_roles (
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    role_id INT REFERENCES roles(id) ON DELETE CASCADE,
    PRIMARY KEY (user_id, role_id)
);

--courses tables-----------------
CREATE TABLE courses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    instructor_id UUID REFERENCES users(id),
    title VARCHAR(200) NOT NULL,
    description TEXT,
    category VARCHAR(80),
    difficulty VARCHAR(20),
    thumbnail_url TEXT,
    price NUMERIC(10,2) DEFAULT 0,
    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMPTZ DEFAULT now()
);
--modules tables--------------------------
CREATE TABLE modules (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    title VARCHAR(200) NOT NULL,
    order_index INT NOT NULL
);
--lectures tables---------------------------
CREATE TABLE lectures (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    module_id UUID REFERENCES modules(id) ON DELETE CASCADE,
    title VARCHAR(200) NOT NULL,
    video_url TEXT,
    transcript TEXT,
    duration_seconds INT,
    order_index INT NOT NULL,
    resource_urls TEXT[]
);

type MigrationFile = {
  filename: string;
  sql: string;
  checksum: string;
};

type AppliedMigrationRow = {
  filename: string;
  checksum: string;
};

type TableRow = {
  table_name: string;
};

const checksum = (content: string): string =>
  createHash("sha256").update(content).digest("hex");

const loadMigrations = async (): Promise<MigrationFile[]> => {
  const files = (await readdir(migrationsDir))
    .filter((name) => /^\d{3}_.+\.sql$/.test(name))
    .sort();

  const migrations: MigrationFile[] = [];

  for (const filename of files) {
    const sql = await readFile(path.join(migrationsDir, filename), "utf8");

    migrations.push({
      filename,
      sql,
      checksum: checksum(sql),
    });
  }

  return migrations;
};

const ensureTrackingTable = async (client: PoolClient): Promise<void> => {
  await client.query(`
    CREATE TABLE IF NOT EXISTS schema_migrations (
      filename TEXT PRIMARY KEY,
      checksum TEXT NOT NULL,
      applied_at TIMESTAMPTZ NOT NULL DEFAULT now()
    )
  `);
};

const verifyBaselineSchema = async (client: PoolClient): Promise<void> => {
  const result = await client.query<TableRow>(
    `
      SELECT table_name
      FROM information_schema.tables
      WHERE table_schema = 'public'
        AND table_type = 'BASE TABLE'
        AND table_name = ANY($1::text[])
    `,
    [REQUIRED_BASELINE_TABLES],
  );

  const found = new Set(result.rows.map((row) => row.table_name));

  const missing = REQUIRED_BASELINE_TABLES.filter((table) => !found.has(table));

  if (missing.length > 0) {
    throw new Error(
      `Baseline stopped. Missing required tables: ${missing.join(", ")}`,
    );
  }
};

const baseline = async (
  client: PoolClient,
  migrations: MigrationFile[],
): Promise<void> => {
  const existing = await client.query<{ count: number }>(
    "SELECT COUNT(*)::int AS count FROM schema_migrations",
  );

  if (existing.rows[0].count > 0) {
    throw new Error(
      "Baseline stopped: schema_migrations already contains records.",
    );
  }

  await verifyBaselineSchema(client);

  const byName = new Map(
    migrations.map((migration) => [migration.filename, migration]),
  );

  for (const filename of BASELINE_FILES) {
    const migration = byName.get(filename);

    if (!migration) {
      throw new Error(`Baseline stopped: missing file ${filename}`);
    }

    await client.query(
      `
        INSERT INTO schema_migrations (filename, checksum)
        VALUES ($1, $2)
      `,
      [migration.filename, migration.checksum],
    );
  }

  console.log(`Baseline recorded ${BASELINE_FILES.length} migrations.`);

  console.log(
    "Existing database schema was verified. Migration SQL was not executed.",
  );
};

const migrate = async (
  client: PoolClient,
  migrations: MigrationFile[],
): Promise<void> => {
  const appliedResult = await client.query<AppliedMigrationRow>(
    "SELECT filename, checksum FROM schema_migrations",
  );

  const applied = new Map<string, string>(
    appliedResult.rows.map((row) => [row.filename, row.checksum]),
  );

  for (const migration of migrations) {
    const previousChecksum = applied.get(migration.filename);

    if (previousChecksum !== undefined) {
      if (previousChecksum !== migration.checksum) {
        throw new Error(
          `Checksum mismatch for ${migration.filename}. ` +
            "Do not edit an applied migration; create a new migration.",
        );
      }

      continue;
    }

    await client.query("BEGIN");

    try {
      await client.query(migration.sql);

      await client.query(
        `
          INSERT INTO schema_migrations (filename, checksum)
          VALUES ($1, $2)
        `,
        [migration.filename, migration.checksum],
      );

      await client.query("COMMIT");

      applied.set(migration.filename, migration.checksum);

      console.log(`Applied: ${migration.filename}`);
    } catch (error) {
      await client.query("ROLLBACK");
      throw error;
    }
  }

  console.log("Migration run completed.");
};

const main = async (): Promise<void> => {
  const command = process.argv[2];

  if (command !== "baseline" && command !== "up") {
    throw new Error("Usage: npm run db:baseline OR npm run db:migrate");
  }

  const client = await pool.connect();
  let lockAcquired = false;

  try {
    await client.query("SELECT pg_advisory_lock($1)", [LOCK_ID]);

    lockAcquired = true;

    await ensureTrackingTable(client);

    const migrations = await loadMigrations();

    if (command === "baseline") {
      await client.query("BEGIN");

      try {
        await baseline(client, migrations);
        await client.query("COMMIT");
      } catch (error) {
        await client.query("ROLLBACK");
        throw error;
      }
    } else {
      await migrate(client, migrations);
    }
  } finally {
    try {
      if (lockAcquired) {
        await client.query("SELECT pg_advisory_unlock($1)", [LOCK_ID]);
      }
    } finally {
      client.release();
      await pool.end();
    }
  }
};

main().catch((error: unknown) => {
  console.error("Migration command failed:", error);
  process.exitCode = 1;
});

import type { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import { pool } from "../config/db.js";

export interface AuthenticatedRequest extends Request {
  user?: {
    id: string;
    roles: string[];
  };
}

export const authenticateToken = async (
  req: Request,
  res: Response,
  next: NextFunction,
): Promise<void> => {
  const authHeader = req.headers.authorization;

  const token = authHeader?.startsWith("Bearer ")
    ? authHeader.slice(7)
    : undefined;

  if (!token) {
    res.status(401).json({
      success: false,
      message: "Access token required",
    });
    return;
  }

  const secret = process.env.JWT_ACCESS_SECRET;

  if (!secret) {
    res.status(500).json({
      success: false,
      message: "Authentication configuration error",
    });
    return;
  }

  try {
    const payload = jwt.verify(token, secret);

    if (
      typeof payload === "string" ||
      typeof payload.sub !== "string" ||
      !Array.isArray(payload.roles) ||
      !payload.roles.every((role) => typeof role === "string")
    ) {
      res.status(401).json({
        success: false,
        message: "Invalid access token",
      });
      return;
    }

    // Check whether the user account is active.
    const userResult = await pool.query(
      "SELECT is_active FROM users WHERE id = $1",
      [payload.sub],
    );

    if (userResult.rowCount === 0 || !userResult.rows[0].is_active) {
      res.status(403).json({
        success: false,
        message: "Account is suspended or unavailable",
      });
      return;
    }

    // Set authenticated user only after the database check.
    (req as AuthenticatedRequest).user = {
      id: payload.sub,
      roles: payload.roles,
    };

    next();
  } catch {
    res.status(401).json({
      success: false,
      message: "Invalid or expired access token",
    });
  }
};

export interface AdminUser {

  full_name: string;
  email: string;
  is_active: boolean;
  created_at: Date;
  roles: string[];

export const getAdminUsers = async (search?: string): Promise<AdminUser[]> => {
  const query = `
    SELECT
      u.id,
      u.full_name,
      u.email,
      u.is_active,
      u.created_at,
      COALESCE(
        ARRAY_AGG(r.name) FILTER (WHERE r.name IS NOT NULL),
        ARRAY[]::VARCHAR[]
      ) AS roles
    FROM users u
    LEFT JOIN user_roles ur ON ur.user_id = u.id
    LEFT JOIN roles r ON r.id = ur.role_id
    WHERE (
      $1::TEXT IS NULL
      OR u.full_name ILIKE '%' || $1 || '%'
      OR u.email ILIKE '%' || $1 || '%'
    )
    GROUP BY u.id
    ORDER BY u.created_at DESC;
  `;

  const result = await pool.query(query, [search?.trim() || null]);

  return result.rows as AdminUser[];
};

export type AdminRole = "student" | "instructor" | "admin";
export type RoleAction = "assign" | "revoke";

export const updateUserRole = async (
  userId: string,
  role: AdminRole,
  action: RoleAction,
): Promise<void> => {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const userResult = await client.query(
      "SELECT id FROM users WHERE id = $1 FOR UPDATE",
      [userId],
    );

    if (userResult.rowCount === 0) {
      throw new Error("USER_NOT_FOUND");
    }

    const roleResult = await client.query(
      "SELECT id FROM roles WHERE name = $1",
      [role],
    );

    if (roleResult.rowCount === 0) {
      throw new Error("ROLE_NOT_FOUND");
    }

    const roleId = roleResult.rows[0].id;

    if (action === "assign") {
      await client.query(
        `INSERT INTO user_roles (user_id, role_id)
         VALUES ($1, $2)
         ON CONFLICT (user_id, role_id) DO NOTHING`,
        [userId, roleId],
      );
    } else {
      if (role === "admin") {
        const adminCountResult = await client.query(
          `SELECT COUNT(*)::int AS count
           FROM user_roles ur
           JOIN roles r ON r.id = ur.role_id
           WHERE r.name = 'admin'`,
        );

        if (adminCountResult.rows[0].count <= 1) {
          throw new Error("CANNOT_REVOKE_LAST_ADMIN");
        }
      }

      await client.query(
        "DELETE FROM user_roles WHERE user_id = $1 AND role_id = $2",
        [userId, roleId],
      );
    }

    await client.query("COMMIT");
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

type RegisterRole = "student" | "instructor";

interface RegisterInput {

export const registerUser = async (input: RegisterInput) => {
  const fullName = input.fullName.trim();
  const email = input.email.trim().toLowerCase();
  const password = input.password;
  const role = input.role ?? "student";

  if (!fullName || !email || !password) {
    throw new Error("Full name, email, and password are required");
  }

  if (password.length < 8) {
    throw new Error("Password must be at least 8 characters");
  }

  if (role !== "student" && role !== "instructor") {
    throw new Error("Invalid registration role");
  }

  const existingUser = await pool.query(
    "SELECT id FROM users WHERE email = $1",
    [email],
  );

  if (existingUser.rowCount && existingUser.rowCount > 0) {
    throw new Error("Email is already registered");
  }

  const passwordHash = await bcrypt.hash(password, 12);
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const userResult = await client.query(
      `INSERT INTO users (full_name, email, password_hash)
       VALUES ($1, $2, $3)
       RETURNING id, full_name, email, created_at`,
      [fullName, email, passwordHash],
    );

    const roleResult = await client.query(
      "SELECT id FROM roles WHERE name = $1",
      [role],
    );

    if (roleResult.rowCount !== 1) {
      throw new Error(`Role "${role}" is not configured`);
    }

    await client.query(
      `INSERT INTO user_roles (user_id, role_id)
       VALUES ($1, $2)`,
      [userResult.rows[0].id, roleResult.rows[0].id],
    );

    await client.query("COMMIT");

    return {
      ...userResult.rows[0],
      roles: [role],
    };
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

export const loginUser = async (emailInput: string, password: string) => {
  const email = emailInput.trim().toLowerCase();

  if (!email || !password) {
    throw new Error("Email and password are required");
  }

  const result = await pool.query(
    `SELECT u.id, u.full_name, u.email, u.password_hash,
            ARRAY_AGG(r.name) AS roles
     FROM users u
     JOIN user_roles ur ON ur.user_id = u.id
     JOIN roles r ON r.id = ur.role_id
     WHERE u.email = $1 AND u.is_active = TRUE
     GROUP BY u.id`,
    [email],
  );

  if (result.rowCount !== 1) {
    throw new Error("Invalid email or password");
  }

  const user = result.rows[0];
  const passwordMatches = await bcrypt.compare(password, user.password_hash);

  if (!passwordMatches) {
    throw new Error("Invalid email or password");
  }

  const secret = process.env.JWT_ACCESS_SECRET;

  if (!secret || secret === "replace_with_a_long_random_secret") {
    throw new Error("JWT access secret is not configured");
  }

  const accessToken = jwt.sign(
    {
      sub: user.id,
      roles: user.roles,
    },
    secret,
    {
      expiresIn: (process.env.JWT_ACCESS_EXPIRES_IN ||
        "15m") as jwt.SignOptions["expiresIn"],
    },
  );
  const refreshToken = await createRefreshToken(user.id);

  return {
    user: {
      id: user.id,
      full_name: user.full_name,
      email: user.email,
      roles: user.roles,
    },
    accessToken,
    refreshToken,
  };
};

const hashRefreshToken = (token: string): string =>
  createHash("sha256").update(token).digest("hex");

const createRefreshToken = async (userId: string) => {
  const secret = process.env.JWT_REFRESH_SECRET;

  if (!secret || secret === "replace_with_a_different_long_random_secret") {
    throw new Error("JWT refresh secret is not configured");
  }

  const token = randomBytes(48).toString("hex");
  const tokenHash = hashRefreshToken(token);

  const expiresAt = new Date();
  expiresAt.setDate(expiresAt.getDate() + 7);

  await pool.query(
    `INSERT INTO refresh_tokens (user_id, token_hash, expires_at)
     VALUES ($1, $2, $3)`,
    [userId, tokenHash, expiresAt],
  );

  return token;
};

export const refreshAccessToken = async (token: string) => {
  if (!token || token.length > 256) {
    throw new Error("Invalid refresh token");
  }

  const tokenHash = hashRefreshToken(token);

  const result = await pool.query(
    `SELECT u.id, ARRAY_AGG(r.name) AS roles
     FROM refresh_tokens rt
     JOIN users u ON u.id = rt.user_id
     JOIN user_roles ur ON ur.user_id = u.id
     JOIN roles r ON r.id = ur.role_id
     WHERE rt.token_hash = $1
       AND rt.expires_at > now()
       AND rt.revoked_at IS NULL
       AND u.is_active = TRUE
     GROUP BY u.id`,
    [tokenHash],
  );

  if (result.rowCount !== 1) {
    throw new Error("Invalid or expired refresh token");
  }

  const user = result.rows[0];
  const secret = process.env.JWT_ACCESS_SECRET;

  if (!secret) {
    throw new Error("JWT access secret is not configured");
  }

  const accessToken = jwt.sign(
    {
      sub: user.id,
      roles: user.roles,
    },
    secret,
    {
      expiresIn: (process.env.JWT_ACCESS_EXPIRES_IN ||
        "15m") as jwt.SignOptions["expiresIn"],
    },
  );

  return { accessToken };
};

export const logoutUser = async (token: string) => {
  if (!token || token.length > 256) {
    throw new Error("Refresh token is required");
  }

  const tokenHash = hashRefreshToken(token);

  const result = await pool.query(
    `UPDATE refresh_tokens
     SET revoked_at = now()
     WHERE token_hash = $1
       AND revoked_at IS NULL
     RETURNING id`,
    [tokenHash],
  );

  return (result.rowCount ?? 0) > 0;
};

interface CreateBookmarkInput {
  userId: string;
  lectureId: string;
  timestampSeconds: number;
}

export async function createLectureBookmark({
  userId,
  lectureId,
  timestampSeconds,
}: CreateBookmarkInput) {
  if (!Number.isInteger(timestampSeconds) || timestampSeconds < 0) {
    throw new Error("Timestamp must be a non-negative integer");
  }

  const lectureResult = await pool.query(
    `SELECT l.id
     FROM lectures l
     JOIN modules m ON m.id = l.module_id
     JOIN courses c ON c.id = m.course_id
     JOIN enrollments e ON e.course_id = c.id
     WHERE l.id = $1
       AND e.student_id = $2
       AND c.status = 'approved'
     LIMIT 1`,
    [lectureId, userId],
  );

  if (lectureResult.rows.length === 0) {
    throw new Error("Lecture not found or you are not enrolled in this course");
  }

  const result = await pool.query(
    `INSERT INTO bookmarks
       (user_id, lecture_id, timestamp_seconds)
     VALUES ($1, $2, $3)
     RETURNING id, user_id, lecture_id,
               timestamp_seconds, created_at`,
    [userId, lectureId, timestampSeconds],
  );

  return result.rows[0];
}

export async function getLectureBookmarks(userId: string, lectureId: string) {
  const result = await pool.query(
    `SELECT
       b.id,
       b.user_id,
       b.lecture_id,
       b.timestamp_seconds,
       b.created_at
     FROM bookmarks b
     JOIN lectures l ON l.id = b.lecture_id
     JOIN modules m ON m.id = l.module_id
     JOIN courses c ON c.id = m.course_id
     JOIN enrollments e ON e.course_id = c.id
     WHERE b.user_id = $1
       AND b.lecture_id = $2
       AND e.student_id = $1
       AND c.status = 'approved'
     ORDER BY b.timestamp_seconds ASC, b.created_at ASC`,

  );

}

export const createCourseController = async (
  req: AuthenticatedRequest,
  res: Response,
) => {
  try {
    if (!req.user) {
      return res.status(401).json({
        success: false,
        message: "Authentication required",
      });
    }

    const userId = req.user.id;

    if (!userId) {
      return res.status(401).json({
        success: false,
        message: "Authentication required",
      });
    }

    const price = Number(req.body.price ?? 0);

    if (!Number.isFinite(price) || price < 0) {
      return res.status(400).json({
        success: false,
        message: "Course price must be a non-negative number",
      });
    }

    const course = await createCourse(userId, {
      ...req.body,
      price,
    });

    let thumbnailUrl: string | undefined;

    if (req.file) {
      thumbnailUrl = await uploadCourseThumbnail(course.id, req.file);

      const updatedCourse = await updateCourse(course.id, userId, {
        thumbnail_url: thumbnailUrl,
      });

      if (!updatedCourse) {
        throw new Error("Failed to save course thumbnail");
      }

      return res.status(201).json({
        success: true,
        data: updatedCourse,
      });
    }

    return res.status(201).json({
      success: true,
      data: course,
    });
  } catch (error) {
    const message =
      error instanceof Error ? error.message : "Failed to create course";

    return res.status(400).json({
      success: false,
      message,
    });
  }
};

export const uploadLectureVideoController = async (
  req: AuthenticatedRequest,
  res: Response,
) => {
  try {
    if (!req.user) {
      return res.status(401).json({
        success: false,
        message: "Authentication required",
      });
    }

    const moduleId = Array.isArray(req.params.id)
      ? req.params.id[0]
      : req.params.id;

    const file = req.file;

    if (!moduleId || !file) {
      return res.status(400).json({
        success: false,
        message: "Module ID and video file are required",
      });
    }

    const { title, transcript, duration_seconds, order_index, resource_urls } =
      req.body ?? {};

    if (
      typeof title !== "string" ||
      !title.trim() ||
      !Number.isInteger(Number(order_index)) ||
      Number(order_index) < 1
    ) {
      return res.status(400).json({
        success: false,
        message: "A lecture title and positive order_index are required",
      });
    }

    const lecture = await createModuleLecture(moduleId, req.user.id, {
      title: title.trim(),
      order_index: Number(order_index),
      transcript,
      duration_seconds:
        duration_seconds === undefined || duration_seconds === ""
          ? undefined
          : Number(duration_seconds),
      resource_urls:
        typeof resource_urls === "string"
          ? JSON.parse(resource_urls)
          : resource_urls,
    });

    if (!lecture) {
      return res.status(404).json({
        success: false,
        message: "Module not found or you are not the course owner",
      });
    }

    const videoKey = await uploadLectureVideo(lecture.id, file);

    const result = await pool.query(
      `UPDATE lectures
       SET video_url = $1
       WHERE id = $2 AND module_id = $3
       RETURNING id, module_id, title, video_url,
                 duration_seconds, order_index, resource_urls`,
      [videoKey, lecture.id, moduleId],
    );

    return res.status(201).json({
      success: true,
      message: "Lecture created and video uploaded successfully",
      data: result.rows[0],
    });
  } catch (error) {
    console.error("Lecture video upload error:", error);

export const deleteModuleController = async (
  req: AuthenticatedRequest,
  res: Response,

        success: false,
        message: "Authentication required",
      });
    }

    const moduleId = Array.isArray(req.params.id)
      ? req.params.id[0]
      : req.params.id;

    if (!moduleId) {
      return res.status(400).json({
        success: false,
        message: "Module ID is required",
      });
    }

    const result = await deleteCourseModule(moduleId, req.user.id);

    if (result.status === "not_found") {
      return res.status(404).json({
        success: false,
        message: "Module not found or you are not the course owner",
      });
    }

    if (result.status === "has_quizzes") {
      return res.status(409).json({
        success: false,
        message: "Cannot delete this module because quizzes are linked to it",
      });
    }

    if (result.status === "has_bookmarks") {
      return res.status(409).json({
        success: false,
        message: "Cannot delete this module because lecture bookmarks exist",
      });
    }

    //MINIO clean

    const cleanupResults = await Promise.allSettled(
      result.videos
        .filter((video): video is typeof video & { video_url: string } =>
          Boolean(video.video_url),
        )
        .map((video) => deleteLectureVideo(video.video_url)),
    );

    const failedCleanupCount = cleanupResults.filter(
      (item) => item.status === "rejected",
    ).length;

    if (failedCleanupCount > 0) {
      console.error(
        `Module ${moduleId} deleted, but ${failedCleanupCount} video object(s) could not be removed from MinIO.`,
        cleanupResults.filter((item) => item.status === "rejected"),
      );
    }

    return res.status(200).json({
      success: true,
      message: "Module deleted successfully",
    });
  } catch (error) {
    const message =
      error instanceof Error ? error.message : "Module deletion failed";

    return res.status(500).json({
      success: false,
      message,
    });
  }
};

export const uploadLectureResourceController = async (
  req: AuthenticatedRequest,
  res: Response,
) => {
  try {
    const instructorId = req.user?.id;
    const lectureId = Array.isArray(req.params.id)
      ? req.params.id[0]
      : req.params.id;
    const file = req.file;

    if (!instructorId) {
      return res.status(401).json({
        success: false,
        message: "Authentication required",
      });
    }

    if (!lectureId || !file) {
      return res.status(400).json({
        success: false,
        message: "Lecture ID and resource file are required",
      });
    }

    // Verify that the lecture belongs to this instructor.
    const ownership = await pool.query(
      `SELECT l.id, l.resource_urls
       FROM lectures l
       JOIN modules m ON m.id = l.module_id
       JOIN courses c ON c.id = m.course_id
       WHERE l.id = $1 AND c.instructor_id = $2
       LIMIT 1`,
      [lectureId, instructorId],
    );

    if (ownership.rowCount === 0) {
      return res.status(404).json({
        success: false,
        message: "Lecture not found or you do not own this course",
      });
    }

    const objectKey = await uploadLectureResource(lectureId, file);
    const existingResources: string[] = ownership.rows[0].resource_urls ?? [];

    const updated = await pool.query(
      `UPDATE lectures
       SET resource_urls = array_append(
         COALESCE(resource_urls, ARRAY[]::TEXT[]),
         $1
       )
       WHERE id = $2
       RETURNING id, module_id, title, resource_urls`,
      [objectKey, lectureId],
    );

    return res.status(201).json({
      success: true,
      message: "Lecture resource uploaded successfully",
      data: {
        ...updated.rows[0],
        resource_count: existingResources.length + 1,
      },
    });
  } catch (error) {
    console.error("Lecture resource upload error:", error);

    return res.status(500).json({
      success: false,
      message: "Failed to upload lecture resource",
    });
  }
};

export const downloadLectureResourceController = async (
  req: AuthenticatedRequest,
  res: Response,
) => {
  try {
    const studentId = req.user?.id;
    const lectureId = Array.isArray(req.params.id)
      ? req.params.id[0]
      : req.params.id;
    const resourceKey = req.query.key;

    if (!studentId) {
      return res.status(401).json({
        success: false,
        message: "Authentication required",
      });
    }

    if (
      !lectureId ||
      typeof resourceKey !== "string" ||
      !resourceKey.startsWith(`lectures/${lectureId}/resources/`)
    ) {
      return res.status(400).json({
        success: false,
        message: "Valid lecture ID and resource key are required",
      });
    }

    const result = await pool.query(
      `SELECT l.id
       FROM lectures l
       JOIN modules m ON m.id = l.module_id
       JOIN courses c ON c.id = m.course_id
       JOIN enrollments e ON e.course_id = c.id
       WHERE l.id = $1
         AND $2 = ANY(l.resource_urls)
         AND e.student_id = $3
         AND c.status = 'approved'
       LIMIT 1`,
      [lectureId, resourceKey, studentId],
    );

    if (result.rowCount === 0) {
      return res.status(403).json({
        success: false,
        message: "Resource not found or you are not enrolled in this course",
      });
    }

    const url = await getLectureResourceUrl(resourceKey);

    return res.status(200).json({
      success: true,
      data: { url, expires_in: 300 },
    });
  } catch (error) {
    console.error("Lecture resource download error:", error);

    return res.status(500).json({
      success: false,
      message: "Failed to generate resource download URL",
    });
  }
};

interface CourseFilters {
  category?: string;
  difficulty?: string;
  q?: string;
  page?: string;
}

export const getPublishedCourses = async (filters: CourseFilters) => {
  const page = Math.max(1, Number.parseInt(filters.page || "1", 10) || 1);
  const limit = 10;
  const offset = (page - 1) * limit;

  const conditions: string[] = ["status = $1"];
  const values: unknown[] = ["approved"];

  if (filters.category?.trim()) {
    values.push(filters.category.trim());
    conditions.push(`category = $${values.length}`);
  }

  if (filters.difficulty?.trim()) {
    values.push(filters.difficulty.trim());
    conditions.push(`LOWER(difficulty) = LOWER($${values.length})`);
  }

  if (filters.q?.trim()) {
    values.push(`%${filters.q.trim()}%`);
    conditions.push(
      `(title ILIKE $${values.length} OR description ILIKE $${values.length})`,
    );
  }

  const whereClause = conditions.join(" AND ");

  const countResult = await pool.query(
    `SELECT COUNT(*)::int AS total
     FROM courses
     WHERE ${whereClause}`,
    values,
  );

  const listValues = [...values, limit, offset];

  const coursesResult = await pool.query(
    `SELECT id, title, description, category, difficulty,
            thumbnail_url, price, instructor_id, created_at
     FROM courses
     WHERE ${whereClause}
     ORDER BY created_at DESC
     LIMIT $${values.length + 1}
     OFFSET $${values.length + 2}`,
    listValues,
  );

  return {
    courses: await Promise.all(
      coursesResult.rows.map(async (course) => ({
        ...course,
        thumbnail_url: course.thumbnail_url
          ? await getCourseThumbnailUrl(course.thumbnail_url)
          : null,
      })),
    ),
    pagination: {
      page,
      limit,
      total: countResult.rows[0].total,
      totalPages: Math.ceil(countResult.rows[0].total / limit),
    },
  };
};

export const getCourseById = async (courseId: string) => {

    `SELECT id, instructor_id, title, description, category,
            difficulty, thumbnail_url, price, status, created_at
     FROM courses
     WHERE id = $1 AND status = 'approved'`,
    [courseId],
  );

  if (courseResult.rowCount !== 1) {
    return null;
  }

  const course = courseResult.rows[0];

  const modulesResult = await pool.query(
    `SELECT id, course_id, title, order_index
     FROM modules
     WHERE course_id = $1
     ORDER BY order_index ASC`,
    [courseId],
  );

  let lectures: Array<{
    id: string;
    module_id: string;
    title: string;
    video_url: string | null;
    duration_seconds: number | null;
    order_index: number;
    resource_urls: string[] | null;
  }> = [];

  if (moduleIds.length > 0) {
    const lecturesResult = await pool.query(
      `SELECT id, module_id, title, video_url,
              duration_seconds, order_index, resource_urls
       FROM lectures
       WHERE module_id = ANY($1::uuid[])
       ORDER BY order_index ASC`,

  }));

  return {
    ...course,
    modules,
  };

interface UpdateCourseInput {
  title?: string;
  description?: string;
  category?: string;
  difficulty?: string;
  thumbnail_url?: string;
  price?: number;
}

export const updateCourse = async (
  courseId: string,
  instructorId: string,
  input: UpdateCourseInput,
) => {
  const allowedFields = [
    "title",
    "description",
    "category",
    "difficulty",
    "thumbnail_url",
    "price",
  ] as const;

  const updates: string[] = [];
  const values: unknown[] = [];

  for (const field of allowedFields) {
    if (input[field] !== undefined) {
      const value = input[field];

      if (field === "title" && (typeof value !== "string" || !value.trim())) {
        throw new Error("Course title cannot be empty");
      }

      if (
        field === "price" &&
        (typeof value !== "number" || !Number.isFinite(value) || value < 0)
      ) {
        throw new Error("Course price must be a non-negative number");
      }

      values.push(
        field === "title" && typeof value === "string" ? value.trim() : value,
      );

      updates.push(`${field} = $${values.length}`);
    }
  }

  if (updates.length === 0) {
    throw new Error("At least one field is required to update");
  }

  values.push(courseId, instructorId);

  const result = await pool.query(
    `UPDATE courses
     SET ${updates.join(", ")}
     WHERE id = $${values.length - 1}
       AND instructor_id = $${values.length}
     RETURNING id, instructor_id, title, description, category,
               difficulty, thumbnail_url, price, status, created_at`,
    values,
  );

  return result.rows[0] ?? null;
};

          ? await getLectureVideoUrl(lecture.video_url)
          : null,
      })),
    );
  }

  const modules = modulesResult.rows.map((module) => ({
    ...module,
    lectures: lectures.filter((lecture) => lecture.module_id === module.id),
  }));

  return {
    ...course,
    modules,
  };
};

export const deleteCourseModule = async (
  moduleId: string,
  instructorId: string,
) => {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const moduleResult = await client.query(
      `SELECT m.id
       FROM modules AS m
       JOIN courses AS c ON c.id = m.course_id
       WHERE m.id = $1
         AND c.instructor_id = $2
         AND c.status = 'approved'
       FOR UPDATE OF m`,
      [moduleId, instructorId],
    );

    if (moduleResult.rowCount === 0) {
      await client.query("ROLLBACK");
      return { status: "not_found" as const };
    }

    const lectureResult = await client.query(
      `SELECT l.id, l.video_url
       FROM lectures AS l
       WHERE l.module_id = $1`,
      [moduleId],
    );

    const lectureIds = lectureResult.rows.map(
      (lecture: { id: string }) => lecture.id,
    );

    const quizResult = await client.query(
      `SELECT id
       FROM quizzes
       WHERE module_id = $1
          OR generated_from_lecture_id = ANY($2::uuid[])
       LIMIT 1`,
      [moduleId, lectureIds],
    );

    if (quizResult.rowCount && quizResult.rowCount > 0) {
      await client.query("ROLLBACK");
      return { status: "has_quizzes" as const };
    }

    const bookmarkResult = await client.query(
      `SELECT id
       FROM bookmarks
       WHERE lecture_id = ANY($1::uuid[])
       LIMIT 1`,
      [lectureIds],
    );

    if (bookmarkResult.rowCount && bookmarkResult.rowCount > 0) {
      await client.query("ROLLBACK");
      return { status: "has_bookmarks" as const };
    }

    await client.query(`DELETE FROM modules WHERE id = $1`, [moduleId]);

    await client.query("COMMIT");

    return {
      status: "deleted" as const,
      videos: lectureResult.rows as Array<{
        id: string;
        video_url: string | null;
      }>,
    };
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

  lectureId: string,
  file: Express.Multer.File,
): Promise<string> {
  const extension = path.extname(file.originalname).toLowerCase();

  const allowedExtensions = [".mp4", ".webm", ".mov", ".m4v"];

  if (!allowedExtensions.includes(extension)) {
    throw new Error("Unsupported video file extension");
  }

  if (!file.mimetype.startsWith("video/")) {
    throw new Error("Only video files are allowed");
  }

  const objectKey = `lectures/${lectureId}/${randomUUID()}${extension}`;

  await s3Client.send(
    new PutObjectCommand({
      Bucket: videoBucket,
      Key: objectKey,
      Body: file.buffer,
      ContentType: file.mimetype,
      ContentLength: file.size,
    }),
  );

  return objectKey;
}

interface UpdateLectureProgressInput {
  watched_seconds: number;
  is_completed: boolean;
}

export async function updateLectureProgress(
  studentId: string,
  lectureId: string,
  input: UpdateLectureProgressInput,
) {
  // Validate watched seconds
  if (!Number.isInteger(input.watched_seconds) || input.watched_seconds < 0) {
    throw new Error("watched_seconds must be a non-negative integer");
  }

  // Validate completion status
  if (typeof input.is_completed !== "boolean") {
    throw new Error("is_completed must be a boolean");
  }

  // Verify lecture access and course enrollment
  const lectureResult = await pool.query(
    `SELECT l.id
     FROM lectures l
     JOIN modules m ON m.id = l.module_id
     JOIN courses c ON c.id = m.course_id
     JOIN enrollments e ON e.course_id = c.id
     WHERE l.id = $1
       AND e.student_id = $2
       AND c.status = 'approved'`,
    [lectureId, studentId],
  );

  if (lectureResult.rows.length === 0) {
    return null;
  }

  // Insert or update lecture progress
  const result = await pool.query(
    `INSERT INTO lecture_progress (
       student_id,
       lecture_id,
       watched_seconds,
       is_completed
     )
     VALUES ($1, $2, $3, $4)
     ON CONFLICT (student_id, lecture_id)
     DO UPDATE SET
       watched_seconds = GREATEST(
         lecture_progress.watched_seconds,
         EXCLUDED.watched_seconds
       ),
       is_completed =
         lecture_progress.is_completed OR EXCLUDED.is_completed,
       updated_at = now()
     RETURNING
       id,
       student_id,
       lecture_id,
       watched_seconds,
       is_completed,
       updated_at`,
    [studentId, lectureId, input.watched_seconds, input.is_completed],
  );

  // Calculate overall course progress
  const courseProgressResult = await pool.query(
    `SELECT
       m.course_id,
       COUNT(l.id)::int AS total_lectures,
       COUNT(l.id) FILTER (
         WHERE lp.is_completed = TRUE
       )::int AS completed_lectures
     FROM lectures l
     JOIN modules m ON m.id = l.module_id
     LEFT JOIN lecture_progress lp
       ON lp.lecture_id = l.id
       AND lp.student_id = $1
     WHERE m.course_id = (
       SELECT m2.course_id
       FROM lectures l2
       JOIN modules m2 ON m2.id = l2.module_id
       WHERE l2.id = $2
     )
     GROUP BY m.course_id`,
    [studentId, lectureId],
  );

  const courseProgress = courseProgressResult.rows[0];

  if (courseProgress && courseProgress.total_lectures > 0) {
    const progressPercentage = Number(
      (
        (courseProgress.completed_lectures / courseProgress.total_lectures) *
        100
      ).toFixed(2),
    );

    // Update enrollment progress
    await pool.query(
      `UPDATE enrollments
       SET progress_percentage = $3,
           completed_at = CASE
              WHEN $3::numeric = 100
                THEN COALESCE(completed_at, now())
               ELSE NULL
            END
       WHERE student_id = $1
         AND course_id = $2`,
      [studentId, courseProgress.course_id, progressPercentage],
    );

    // Issue certificate when the course is completed
    if (progressPercentage === 100) {
      await issueCourseCertificate(studentId, courseProgress.course_id);
    }
  }

  return result.rows[0];
}

export const createQuizController = async (
  req: Request,
  res: Response,
): Promise<void> => {
  const authenticatedReq = req as AuthenticatedRequest;
  const instructorId = authenticatedReq.user?.id;

  if (!instructorId) {
    res.status(401).json({
      success: false,
      message: "Authentication required",
    });
    return;
  }

  const { module_id, title, questions } = req.body as {
    module_id?: string;
    title?: string;
    questions?: Array<{
      question_text?: string;
      question_type?: string;
      order_index?: number;
      options?: Array<{
        option_text?: string;
        is_correct?: boolean;
      }>;
    }>;
  };

  if (
    typeof module_id !== "string" ||
    typeof title !== "string" ||
    !title.trim() ||
    !Array.isArray(questions) ||
    questions.length === 0
  ) {
    res.status(400).json({
      success: false,
      message: "module_id, title and at least one question are required",
    });
    return;
  }

  for (const question of questions) {
    if (
      typeof question.question_text !== "string" ||
      !question.question_text.trim() ||
      !["mcq", "multi_select", "short_answer"].includes(
        question.question_type ?? "",
      ) ||
      !Number.isInteger(question.order_index) ||
      (question.question_type !== "short_answer" &&
        (!Array.isArray(question.options) ||
          question.options.length < 2 ||
          !question.options.every(
            (option) =>
              typeof option.option_text === "string" &&
              option.option_text.trim() &&
              typeof option.is_correct === "boolean",
          ) ||
          !question.options.some((option) => option.is_correct)))
    ) {
      res.status(400).json({
        success: false,
        message: "Invalid question or options",
      });
      return;
    }
  }

  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const moduleResult = await client.query(
      `SELECT m.id
       FROM modules m
       JOIN courses c ON c.id = m.course_id
       WHERE m.id = $1 AND c.instructor_id = $2`,
      [module_id, instructorId],
    );

    if (moduleResult.rowCount !== 1) {
      await client.query("ROLLBACK");

      res.status(404).json({
        success: false,
        message: "Module not found or not owned by this instructor",
      });
      return;
    }

    const quizResult = await client.query(
      `INSERT INTO quizzes (module_id, title, is_ai_generated)
       VALUES ($1, $2, FALSE)
       RETURNING id, module_id, title, is_ai_generated`,
      [module_id, title.trim()],
    );

    const quiz = quizResult.rows[0];

    for (const question of questions) {
      const questionResult = await client.query(
        `INSERT INTO quiz_questions
           (quiz_id, question_text, question_type, order_index)
         VALUES ($1, $2, $3, $4)
         RETURNING id, question_text, question_type, order_index`,
        [
          quiz.id,
          question.question_text!.trim(),
          question.question_type,
          question.order_index,
        ],
      );

      const savedQuestion = questionResult.rows[0];

      if (question.question_type !== "short_answer") {
        for (const option of question.options!) {
          await client.query(
            `INSERT INTO quiz_options
               (question_id, option_text, is_correct)
             VALUES ($1, $2, $3)`,
            [savedQuestion.id, option.option_text!.trim(), option.is_correct],
          );
        }
      }
    }

    await client.query("COMMIT");

    res.status(201).json({
      success: true,
      message: "Quiz created successfully",
      data: quiz,
    });
  } catch (error) {
    await client.query("ROLLBACK");

    console.error("Create quiz failed:", error);

    res.status(500).json({
      success: false,
      message: "Failed to create quiz",
    });
  } finally {
    client.release();
  }
};

export const submitQuizAttemptController = async (
  req: Request,
  res: Response,
): Promise<void> => {
  const authenticatedReq = req as AuthenticatedRequest;
  const studentId = authenticatedReq.user?.id;

  const attemptId = Array.isArray(req.params.id)
    ? (req.params.id[0] ?? "")
    : (req.params.id ?? "");

  if (!studentId) {
    res.status(401).json({
      success: false,
      message: "Authentication required",
    });
    return;
  }

  if (typeof attemptId !== "string" || !attemptId) {
    res.status(400).json({
      success: false,
      message: "Invalid attempt ID",
    });
    return;
  }

  const { answers } = req.body as {
    answers?: Array<{
      question_id: string;
      selected_option_ids?: string[];
      text_answer?: string;
    }>;
  };

  if (!Array.isArray(answers)) {
    res.status(400).json({
      success: false,
      message: "answers must be an array",
    });
    return;
  }

  try {
    const result = await submitQuizAttempt(attemptId, studentId, answers);

    if (!result) {
      res.status(404).json({
        success: false,
        message: "Open quiz attempt not found",
      });
      return;
    }

    if (result.attempt.score === 100) {
      await awardBadge(studentId, "Perfect Quiz Score");
    }

    try {
      await recommendationQueue.add(
        "recalculate-recommendation-scores",
        { studentId },
        {
          attempts: 3,
          backoff: {
            type: "exponential",
            delay: 1000,
          },
          removeOnComplete: true,
          removeOnFail: false,
        },
      );
    } catch (error) {
      console.error("Failed to enqueue recommendation recalculation:", error);
    }

    res.status(200).json({
      success: true,
      message: "Quiz submitted successfully",
      data: result,
    });
  } catch (error) {
    if (error instanceof Error && error.message === "Invalid quiz answer") {
      res.status(400).json({
        success: false,
        message: error.message,
      });
      return;
    }

    if (
      error instanceof Error &&
      [
        "Short-answer questions cannot have selected options",
        "Objective questions cannot have text answers",
        "Duplicate selected option IDs",
        "Selected option does not belong to question",
      ].includes(error.message)
    ) {
      res.status(400).json({
        success: false,
        message: error.message,
      });
      return;
    }

    console.error("Submit quiz attempt failed:", error);

    res.status(500).json({
      success: false,
      message: "Failed to submit quiz",
    });
  }
};

export const startQuizAttempt = async (quizId: string, studentId: string) => {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const quizResult = await client.query(
      `SELECT q.id
       FROM quizzes q
       JOIN modules m ON m.id = q.module_id
       JOIN courses c ON c.id = m.course_id
       JOIN enrollments e ON e.course_id = c.id
       WHERE q.id = $1
         AND e.student_id = $2
         AND q.approval_status = 'approved'`,
      [quizId, studentId],
    );

    if (quizResult.rowCount !== 1) {
      await client.query("ROLLBACK");
      return null;
    }

    const attemptResult = await client.query(
      `INSERT INTO quiz_attempts (quiz_id, user_id)
       VALUES ($1, $2)
       RETURNING id, quiz_id, user_id, score, started_at, submitted_at`,
      [quizId, studentId],
    );

    await client.query("COMMIT");

    return attemptResult.rows[0];
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

export const getQuizForStudent = async (quizId: string) => {
  const quizResult = await pool.query(
    `SELECT id, title, module_id
     FROM quizzes
     WHERE id = $1
      AND approval_status = 'approved'`,
    [quizId],
  );

  if (quizResult.rowCount !== 1) {
    return null;
  }

  const questionsResult = await pool.query(
    `SELECT id, question_text, question_type, order_index
     FROM quiz_questions
     WHERE quiz_id = $1
     ORDER BY order_index ASC`,
    [quizId],
  );

  const questions = await Promise.all(
    questionsResult.rows.map(async (question) => {
      const optionsResult = await pool.query(
        `SELECT id, option_text
         FROM quiz_options
         WHERE question_id = $1
         ORDER BY id ASC`,
        [question.id],
      );

      return {
        ...question,
        options:
          question.question_type === "short_answer" ? [] : optionsResult.rows,
      };
    }),
  );

  return {
    ...quizResult.rows[0],
    questions,
  };
};

export const submitQuizAttempt = async (
  attemptId: string,
  studentId: string,
  answers: Array<{
    question_id: string;
    selected_option_ids?: string[];
    text_answer?: string;
  }>,
) => {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const attemptResult = await client.query(
      `SELECT id, quiz_id
       FROM quiz_attempts
       WHERE id = $1
         AND user_id = $2
         AND submitted_at IS NULL
       FOR UPDATE`,
      [attemptId, studentId],
    );

    if (attemptResult.rowCount !== 1) {
      await client.query("ROLLBACK");
      return null;
    }

    const { quiz_id: quizId } = attemptResult.rows[0];

    const questionsResult = await client.query(
      `SELECT id, question_type
       FROM quiz_questions
       WHERE quiz_id = $1`,
      [quizId],
    );

    const questions = questionsResult.rows;

    // Every question must have exactly one answer object.
    if (!Array.isArray(answers) || answers.length !== questions.length) {
      throw new Error("Exactly one answer is required for each question");
    }

    const questionMap = new Map(
      questions.map((question) => [question.id, question]),
    );

    const correctAnswers = new Map<string, string[]>();

    const optionsResult = await client.query(
      `SELECT qo.question_id, qo.id
       FROM quiz_options qo
       JOIN quiz_questions qq ON qq.id = qo.question_id
       WHERE qq.quiz_id = $1
         AND qo.is_correct = TRUE`,
      [quizId],
    );

    for (const option of optionsResult.rows) {
      const existing = correctAnswers.get(option.question_id) ?? [];

      existing.push(option.id);
      correctAnswers.set(option.question_id, existing);
    }

    const answerMap = new Map<string, (typeof answers)[number]>();

    for (const answer of answers) {
      if (
        !answer ||
        typeof answer.question_id !== "string" ||
        answerMap.has(answer.question_id) ||
        !questionMap.has(answer.question_id) ||
        (answer.selected_option_ids !== undefined &&
          (!Array.isArray(answer.selected_option_ids) ||
            !answer.selected_option_ids.every(
              (id) => typeof id === "string",
            ))) ||
        (answer.text_answer !== undefined &&
          typeof answer.text_answer !== "string")
      ) {
        throw new Error("Invalid quiz answer");
      }

      answerMap.set(answer.question_id, answer);
    }

    // Defensive check: reject missing or duplicate question answers.
    if (answerMap.size !== questions.length) {
      throw new Error("Exactly one answer is required for each question");
    }

    let correctCount = 0;

    for (const question of questions) {
      const answer = answerMap.get(question.id);

      // The checks above guarantee that each question has an answer.
      if (!answer) {
        throw new Error("Exactly one answer is required for each question");
      }

      const selected = answer.selected_option_ids ?? [];

      if (question.question_type === "short_answer" && selected.length > 0) {
        throw new Error("Short-answer questions cannot have selected options");
      }

      if (
        question.question_type !== "short_answer" &&
        answer.text_answer !== undefined
      ) {
        throw new Error("Objective questions cannot have text answers");
      }

      const uniqueSelected = new Set(selected);

      if (uniqueSelected.size !== selected.length) {
        throw new Error("Duplicate selected option IDs");
      }

      if (selected.length > 0) {
        const validOptions = await client.query(
          `SELECT id
           FROM quiz_options
           WHERE question_id = $1
             AND id = ANY($2::uuid[])`,
          [question.id, selected],
        );

        if (validOptions.rowCount !== selected.length) {
          throw new Error("Selected option does not belong to question");
        }
      }

      let isCorrect: boolean | null = null;

      if (question.question_type !== "short_answer") {
        const expected = correctAnswers.get(question.id) ?? [];

        const expectedSet = new Set(expected);

        isCorrect =
          selected.length === expected.length &&
          selected.every((id) => expectedSet.has(id));

        if (isCorrect) {
          correctCount++;
        }
      }

      await client.query(
        `INSERT INTO quiz_answers
           (attempt_id, question_id, selected_option_ids, text_answer, is_correct)
         VALUES ($1, $2, $3::uuid[], $4, $5)`,
        [
          attemptId,
          question.id,
          selected,
          answer.text_answer ?? null,
          isCorrect,
        ],
      );
    }

    const objectiveQuestions = questions.filter(
      (question) => question.question_type !== "short_answer",
    ).length;

    const score =
      objectiveQuestions === 0
        ? null
        : Number(((correctCount / objectiveQuestions) * 100).toFixed(2));

    const updatedAttempt = await client.query(
      `UPDATE quiz_attempts
       SET score = $2, submitted_at = NOW()
       WHERE id = $1
       RETURNING id, quiz_id, user_id, score, started_at, submitted_at`,
      [attemptId, score],
    );

    await client.query("COMMIT");

    return {
      attempt: updatedAttempt.rows[0],
      total_questions: questions.length,
      objective_questions: objectiveQuestions,
      correct_answers: correctCount,
    };
  } catch (error) {
    // All thrown validation errors roll back the transaction here.
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

export const getCourseRecommendations = async (studentId: string) => {
  const result = await pool.query(
    `
      WITH performance AS (
        SELECT
          LOWER(c.category) AS category,
          LOWER(c.difficulty) AS difficulty,
          AVG(ms.mastery_score)::numeric(5,2) AS average_score
        FROM module_mastery_scores ms
        JOIN modules m ON m.id = ms.module_id
        JOIN courses c ON c.id = m.course_id
        WHERE ms.student_id = $1
          AND ms.scored_attempts > 0
        GROUP BY LOWER(c.category), LOWER(c.difficulty)
      )
      SELECT
        c.id,
        c.title,
        c.description,
        c.category,
        c.difficulty,
        c.thumbnail_url,
        c.price,
        p.average_score AS related_quiz_average,
        CASE
          WHEN p.average_score IS NULL THEN 0
          ELSE 100 - p.average_score
        END AS performance_priority
      FROM courses c
      LEFT JOIN performance p
        ON LOWER(c.category) = p.category
        AND LOWER(c.difficulty) = p.difficulty
      WHERE c.status = 'approved'
        AND NOT EXISTS (
          SELECT 1
          FROM enrollments e
          WHERE e.course_id = c.id
            AND e.student_id = $1
        )
      ORDER BY
        CASE WHEN p.average_score IS NULL THEN 1 ELSE 0 END,
        performance_priority DESC,
        c.created_at DESC
      LIMIT 5
    `,
    [studentId],
  );

  return result.rows;
};

interface CertificateDetails {
  user_id: string;
  course_id: string;
  student_name: string;
  course_title: string;
}

export async function issueCourseCertificate(
  userId: string,
  courseId: string,
): Promise<{ id: string; certificate_url: string } | null> {
  const detailsResult = await pool.query<CertificateDetails>(
    `SELECT
       e.student_id AS user_id,
       c.id AS course_id,
       u.full_name AS student_name,
       c.title AS course_title
     FROM enrollments e
     JOIN courses c ON c.id = e.course_id
     JOIN users u ON u.id = e.student_id
     WHERE e.student_id = $1
       AND e.course_id = $2
       AND e.progress_percentage = 100
       AND e.completed_at IS NOT NULL
       AND c.status = 'approved'`,
    [userId, courseId],
  );

  const details = detailsResult.rows[0];

  if (!details) {
    return null;
  }

  // Return the existing certificate if one has already been issued.
  const existingResult = await pool.query<{
    id: string;
    certificate_url: string;
  }>(
    `SELECT id, certificate_url
     FROM certificates
     WHERE user_id = $1 AND course_id = $2`,
    [userId, courseId],
  );

  if (existingResult.rows[0]?.certificate_url) {
    return existingResult.rows[0];
  }

  const document = new PDFDocument({ size: "A4", layout: "landscape" });
  const chunks: Buffer[] = [];

  const pdfBuffer = await new Promise<Buffer>((resolve, reject) => {
    document.on("data", (chunk: Buffer) => chunks.push(chunk));
    document.on("end", () => resolve(Buffer.concat(chunks)));
    document.on("error", reject);

    document.fontSize(28).text("Certificate of Completion", {
      align: "center",
    });
    document.moveDown(1.5);
    document.fontSize(16).text("This certificate is awarded to", {
      align: "center",
    });
    document.moveDown(0.5);
    document.fontSize(24).text(details.student_name, {
      align: "center",
    });
    document.moveDown(0.5);
    document.fontSize(16).text("for successfully completing", {
      align: "center",
    });
    document.moveDown(0.5);
    document.fontSize(22).text(details.course_title, {
      align: "center",
    });
    document.moveDown(2);
    document
      .fontSize(10)
      .text(`Issued on ${new Date().toISOString().slice(0, 10)}`, {
        align: "center",
      });

    document.end();
  });

  const objectKey = `certificates/${userId}/${courseId}/${randomUUID()}.pdf`;

  await s3Client.send(
    new PutObjectCommand({
      Bucket: videoBucket,
      Key: objectKey,
      Body: pdfBuffer,
      ContentType: "application/pdf",
      ContentLength: pdfBuffer.length,
    }),
  );

  const insertResult = await pool.query<{
    id: string;
    certificate_url: string;
  }>(
    `INSERT INTO certificates (user_id, course_id, certificate_url)
     VALUES ($1, $2, $3)
     ON CONFLICT (user_id, course_id) DO NOTHING
     RETURNING id, certificate_url`,
    [userId, courseId, objectKey],
  );

  if (insertResult.rows[0]) {
    const completedCount = await pool.query<{ count: string }>(
      `SELECT COUNT(*)::text AS count
       FROM certificates
       WHERE user_id = $1`,
      [userId],
    );

    if (Number(completedCount.rows[0]?.count) === 1) {
      await awardBadge(userId, "First Course Completed");
    }

    return insertResult.rows[0];
  }

  const concurrentResult = await pool.query<{
    id: string;
    certificate_url: string;
  }>(
    `SELECT id, certificate_url
     FROM certificates
     WHERE user_id = $1 AND course_id = $2`,
    [userId, courseId],
  );

  return concurrentResult.rows[0] ?? null;
}

export const downloadCertificateController = async (
  req: AuthenticatedRequest,
  res: Response,
): Promise<void> => {
  const userId = req.user?.id;
  const certificateId = req.params.id;

  if (!userId) {
    res.status(401).json({ message: "Unauthorized" });
    return;
  }

  if (!certificateId || Array.isArray(certificateId)) {
    res.status(400).json({ message: "Valid certificate ID is required" });
    return;
  }

  try {
    const certificateService = certificatesService as unknown as {
      getCertificatePdf?: (
        userId: string,
        certificateId: string,
      ) => Promise<{ filename: string; buffer: Buffer } | null>;
    }; 

    if (!certificateService.getCertificatePdf) {
      res.status(500).json({ message: "Failed to download certificate" });
      return;
    }

    const certificate = await certificateService.getCertificatePdf(
      userId,
      certificateId,
    );

    if (!certificate) {
      res.status(404).json({ message: "Certificate not found" });
      return;
    }

    res.setHeader("Content-Type", "application/pdf");
    res.setHeader(
      "Content-Disposition",
      `attachment; filename="${certificate.filename}"`,
    );
    res.setHeader("Content-Length", certificate.buffer.length);

    res.status(200).send(certificate.buffer);
  } catch (error) {
    console.error("Download certificate error:", error);
    res.status(500).json({ message: "Failed to download certificate" });
  }
};

    const updated = await client.query<Streak>(
      `UPDATE streaks
       SET
         current_streak = CASE
           WHEN last_active_date = CURRENT_DATE
             THEN current_streak
           WHEN last_active_date = CURRENT_DATE - 1
             THEN current_streak + 1
           ELSE 1
         END,

    );

interface CreateCheckoutInput {
  userId: string;
  courseId: string;
}

export async function createCheckoutSession({
  userId,
  courseId,
}: CreateCheckoutInput) {
  // Always retrieve the course price from PostgreSQL.
  const courseResult = await pool.query(
    `SELECT id, title, price
     FROM courses
     WHERE id = $1 AND status = 'approved'`,
    [courseId],
  );

  if (courseResult.rows.length === 0) {
    throw new Error("APPROVED_COURSE_NOT_FOUND");
  }

  const course = courseResult.rows[0];
  const amountInPaise = Math.round(Number(course.price) * 100);

  if (!Number.isSafeInteger(amountInPaise) || amountInPaise <= 0) {
    throw new Error("INVALID_COURSE_PRICE");
  }

  // Avoid creating a second payment for an existing enrollment.
  const enrollmentResult = await pool.query(
    `SELECT id
     FROM enrollments
     WHERE student_id = $1 AND course_id = $2`,
    [userId, courseId],
  );

  if (enrollmentResult.rows.length > 0) {
    throw new Error("ALREADY_ENROLLED");
  }

  const paymentResult = await pool.query(
    `INSERT INTO payments
       (user_id, course_id, amount, status)
     VALUES ($1, $2, $3, 'pending')
     RETURNING id`,
    [userId, courseId, course.price],
  );

  const paymentId = paymentResult.rows[0].id;

  try {
    const frontendUrl = process.env.FRONTEND_URL || "http://localhost:5173";

    const session = await stripe.checkout.sessions.create({
      mode: "payment",
      line_items: [
        {
          price_data: {
            currency: "inr",
            product_data: {
              name: course.title,
            },
            unit_amount: amountInPaise,
          },
          quantity: 1,
        },
      ],
      success_url: `${frontendUrl}/payment/success?session_id={CHECKOUT_SESSION_ID}`,
      cancel_url: `${frontendUrl}/payment/cancel`,
      client_reference_id: userId,
      metadata: {
        paymentId,
        userId,
        courseId,
      },
    });

    await pool.query(
      `UPDATE payments
       SET stripe_checkout_session_id = $1
       WHERE id = $2`,
      [session.id, paymentId],
    );

    return {
      paymentId,
      checkoutSessionId: session.id,
      checkoutUrl: session.url,
    };
  } catch (error) {
    await pool.query(
      `UPDATE payments
       SET status = 'failed'
       WHERE id = $1 AND status = 'pending'`,
      [paymentId],
    );

    throw error;
  }
}


import type { Request, Response } from "express";
import Stripe from "stripe";
import { pool } from "../../config/db.js";

const secretKey = process.env.STRIPE_SECRET_KEY;
const webhookSecret = process.env.STRIPE_WEBHOOK_SECRET;

const stripe = secretKey ? new Stripe(secretKey) : null;

export async function handleStripeWebhook(
  req: Request,
  res: Response,
): Promise<void> {
  // Temporary debugging: verify webhook request reaches this handler.
  console.log("WEBHOOK HIT", {
    signaturePresent: typeof req.headers["stripe-signature"] === "string",
    bodyIsBuffer: Buffer.isBuffer(req.body),
  });

  if (!stripe || !webhookSecret) {
    console.error("Stripe webhook configuration is missing");

    res.status(500).json({
      success: false,
      message: "Stripe webhook configuration is missing",
    });
    return;
  }

  const signature = req.headers["stripe-signature"];

  if (typeof signature !== "string" || !Buffer.isBuffer(req.body)) {
    console.error("Webhook request validation failed:", {
      signaturePresent: typeof signature === "string",
      bodyIsBuffer: Buffer.isBuffer(req.body),
    });

    res.status(400).json({
      success: false,
      message: "Invalid Stripe webhook request",
    });
    return;
  }

  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(req.body, signature, webhookSecret);
  } catch (error) {
    console.error(
      "Stripe signature verification failed:",
      error instanceof Error ? error.message : String(error),
    );

    res.status(400).json({
      success: false,
      message: "Invalid Stripe webhook signature",
    });
    return;
  }

  try {
    if (event.type === "checkout.session.completed") {
      const session = event.data.object as Stripe.Checkout.Session;

      // Ignore completed sessions that have not been paid.
      if (session.payment_status !== "paid") {
        res.status(200).json({ received: true });
        return;
      }

      const paymentId = session.metadata?.paymentId;
      const userId = session.metadata?.userId;
      const courseId = session.metadata?.courseId;

      if (
        !paymentId ||
        !userId ||
        !courseId ||
        typeof session.id !== "string"
      ) {
        console.error("Checkout metadata is incomplete", {
          sessionIdPresent: typeof session.id === "string",
          paymentIdPresent: Boolean(paymentId),
          userIdPresent: Boolean(userId),
          courseIdPresent: Boolean(courseId),
        });

        res.status(400).json({
          success: false,
          message: "Checkout metadata is incomplete",
        });
        return;
      }

      const client = await pool.connect();

      try {
        await client.query("BEGIN");

        const paymentResult = await client.query(
          `SELECT id, user_id, course_id, amount, status
           FROM payments
           WHERE id = $1
             AND user_id = $2
             AND course_id = $3
             AND stripe_checkout_session_id = $4
           FOR UPDATE`,
          [paymentId, userId, courseId, session.id],
        );

        if (paymentResult.rows.length !== 1) {
          throw new Error("PAYMENT_RECORD_NOT_FOUND");
        }

        const payment = paymentResult.rows[0];

        if (payment.status !== "success") {
          const amountInPaise = Math.round(Number(payment.amount) * 100);

          if (
            session.currency !== "inr" ||
            session.amount_total !== amountInPaise
          ) {
            throw new Error("PAYMENT_AMOUNT_MISMATCH");
          }

          await client.query(
            `UPDATE payments
             SET status = 'success',
                 stripe_payment_intent_id = $1
             WHERE id = $2`,
            [
              typeof session.payment_intent === "string"
                ? session.payment_intent
                : null,
              paymentId,
            ],
          );
        }

        const courseResult = await client.query(
          `SELECT id
           FROM courses
           WHERE id = $1 AND status = 'approved'`,
          [courseId],
        );

        if (courseResult.rows.length !== 1) {
          throw new Error("COURSE_NOT_AVAILABLE");
        }

        await client.query(
          `INSERT INTO enrollments (student_id, course_id)
           VALUES ($1, $2)
           ON CONFLICT (student_id, course_id) DO NOTHING`,
          [userId, courseId],
        );

        await client.query("COMMIT");
      } catch (error) {
        await client.query("ROLLBACK");
        throw error;
      } finally {
        client.release();
      }
    }

    res.status(200).json({ received: true });
  } catch (error) {
    console.error(
      "Stripe webhook processing error:",
      error instanceof Error ? error.message : String(error),
    );

    res.status(500).json({
      success: false,
      message: "Webhook processing failed",
    });
  }
}

interface LoginResponse {
  success: boolean;
  message: string;
  data?: {
    user: {
      id: string;
      full_name: string;
      email: string;
      roles: string[];
    };
    accessToken: string;
  };
}

interface RegistrationState {
  registrationSuccess?: boolean;
  email?: string;
}

export default function LoginPage() {

  const navigate = useNavigate();

  const [message, setMessage] = useState("");

            "message" in errorData &&
            typeof errorData.message === "string"

            Account created successfully. Please sign in.
          </p>
        )}

            <label htmlFor="email" className="text-sm font-medium">
              Email
            </label>

            <input
              id="email"
              name="email"
              type="email"
              autoComplete="email"
              required
              value={email}
              onChange={(event) => setEmail(event.target.value)}
              className="w-full rounded-md border bg-background px-3 py-2 outline-none focus:ring-2 focus:ring-ring"

        const response = await api.get("/courses", {
          params: {
            q: search.trim() || undefined,
            category: category.trim() || undefined,
            difficulty: difficulty || undefined,
            page,
          },
        });

        const result: CourseCatalogResponse | undefined = response.data?.data;

          throw new Error("Unexpected courses API response.");

          setCourses(result.courses);
          setPagination(result.pagination);

        console.error("Failed to fetch courses:", err);

        if (active) {
          setError("Courses could not be loaded. Please try again.");

      } finally {
        if (active) {
          setLoading(false);

interface ApiResponse<T> {
  data?: T;
  course?: T;
}

function getResponseData<T>(response: { data?: ApiResponse<T> | T }): T | null {
  const body = response.data;

  if (!body || typeof body !== "object") {
    return null;
  }

  if ("data" in body && body.data !== undefined) {
    return body.data as T;
  }

  return body as T;
}

  const { id: courseId } = useParams();

  const videoRef = useRef<HTMLVideoElement>(null);
  const lastSavedSecondRef = useRef(-1);

  const [course, setCourse] = useState<Course | null>(null);
  const [activeLecture, setActiveLecture] = useState<Lecture | null>(null);
  const [progress, setProgress] = useState<Progress | null>(null);
  const [notes, setNotes] = useState<Note[]>([]);
  const [bookmarks, setBookmarks] = useState<Bookmark[]>([]);
  const [noteContent, setNoteContent] = useState("");
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");
  const [downloadingKey, setDownloadingKey] = useState<string | null>(null);

  const loadLectureData = useCallback(async (lecture: Lecture) => {
    setActiveLecture(lecture);
    setProgress(null);
    setNotes([]);
    setBookmarks([]);
    setNoteContent("");
    setError("");
    setMessage("");
    lastSavedSecondRef.current = -1;

    try {
      const [progressResponse, notesResponse, bookmarksResponse] =
        await Promise.all([
          api.get(`/lectures/${lecture.id}/progress`),
          api.get(`/lectures/${lecture.id}/notes`),
          api.get(`/lectures/${lecture.id}/bookmarks`),
        ]);

      const progressData = getResponseData<Progress>(progressResponse);
      const notesData = getResponseData<Note[]>(notesResponse);
      const bookmarksData = getResponseData<Bookmark[]>(bookmarksResponse);

      setProgress(
        progressData && typeof progressData.watched_seconds === "number"
          ? progressData
          : null,
      );

      setNotes(
        Array.isArray(notesData)
          ? [...notesData].sort(
              (a, b) => a.timestamp_seconds - b.timestamp_seconds,
            )
          : [],
      );

      setBookmarks(
        Array.isArray(bookmarksData)
          ? [...bookmarksData].sort(
              (a, b) => a.timestamp_seconds - b.timestamp_seconds,
            )
          : [],
      );
    } catch (err) {
      console.error("Failed to load lecture data:", err);
      setError("Could not load saved progress, notes, or bookmarks.");
    }
  }, []);

  useEffect(() => {
    let cancelled = false;

    async function loadCourse() {
      setLoading(true);
      setError("");
      setMessage("");

      try {
        if (!courseId) {
          throw new Error("Course ID is missing.");
        }

        const response = await api.get(`/courses/${courseId}`);
        const responseBody = response.data;

        const courseData = (responseBody?.data ??
          responseBody?.course ??
          responseBody) as Course;

        if (!courseData || !Array.isArray(courseData.modules)) {
          throw new Error("Unexpected course response.");
        }

        const normalizedCourse: Course = {
          ...courseData,
          modules: courseData.modules.map((module) => ({
            ...module,
            lectures: Array.isArray(module.lectures) ? module.lectures : [],
          })),
        };

        if (cancelled) return;

        setCourse(normalizedCourse);

        const firstLecture = normalizedCourse.modules
          .flatMap((module) => module.lectures)
          .find((lecture) => Boolean(lecture.video_url));

        if (firstLecture) {
          await loadLectureData(firstLecture);
        } else {
          setActiveLecture(null);
          setMessage("This course does not have a video lecture yet.");
        }
      } catch (err) {
        console.error("Failed to load course:", err);

        if (!cancelled) {
          setError("Could not load this course.");
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    void loadCourse();

    return () => {
      cancelled = true;
    };
  }, [courseId, loadLectureData]);

  async function saveProgress(watchedSeconds: number, completed: boolean) {
    if (!activeLecture) return;

    const safeSeconds = Math.max(0, Math.floor(watchedSeconds));

    try {
      await api.post(`/lectures/${activeLecture.id}/progress`, {
        watched_seconds: safeSeconds,
        is_completed: completed,
      });

      setProgress((current) => ({
        watched_seconds: Math.max(current?.watched_seconds ?? 0, safeSeconds),
        is_completed: completed || Boolean(current?.is_completed),
      }));

      setError("");
    } catch (err) {
      console.error("Failed to save lecture progress:", err);
      setError("Could not save lecture progress.");
    }
  }

  async function addNote() {
    if (!activeLecture || !noteContent.trim()) return;

    const timestamp = Math.floor(videoRef.current?.currentTime ?? 0);

    setSaving(true);
    setError("");
    setMessage("");

    try {
      const response = await api.post(`/lectures/${activeLecture.id}/notes`, {
        timestamp_seconds: timestamp,
        content: noteContent.trim(),
      });

      const note = getResponseData<Note>(response);

      if (note) {
        setNotes((current) =>
          [...current, note].sort(
            (a, b) => a.timestamp_seconds - b.timestamp_seconds,
          ),
        );
      } else {
        const notesResponse = await api.get(
          `/lectures/${activeLecture.id}/notes`,
        );

        const notesData = getResponseData<Note[]>(notesResponse);

        if (Array.isArray(notesData)) {
          setNotes(
            [...notesData].sort(
              (a, b) => a.timestamp_seconds - b.timestamp_seconds,
            ),
          );
        }
      }

      setNoteContent("");
      setMessage("Note saved.");
    } catch (err) {
      console.error("Failed to save note:", err);
      setError("Could not save your note.");
    } finally {
      setSaving(false);
    }
  }

  async function addBookmark() {
    if (!activeLecture) return;

    const timestamp = Math.floor(videoRef.current?.currentTime ?? 0);

    setError("");
    setMessage("");

    try {
      const response = await api.post(
        `/lectures/${activeLecture.id}/bookmarks`,
        { timestamp_seconds: timestamp },
      );

      const bookmark = getResponseData<Bookmark>(response);

      if (bookmark) {
        setBookmarks((current) =>
          [...current, bookmark].sort(
            (a, b) => a.timestamp_seconds - b.timestamp_seconds,
          ),
        );
      } else {
        const bookmarksResponse = await api.get(
          `/lectures/${activeLecture.id}/bookmarks`,
        );

        const bookmarksData = getResponseData<Bookmark[]>(bookmarksResponse);

        if (Array.isArray(bookmarksData)) {
          setBookmarks(
            [...bookmarksData].sort(
              (a, b) => a.timestamp_seconds - b.timestamp_seconds,
            ),
          );
        }
      }

      setMessage("Bookmark saved.");
    } catch (err) {
      console.error("Failed to save bookmark:", err);
      setError("Could not save bookmark.");
    }
  }

  async function downloadResource(lectureId: string, resourceKey: string) {
    if (downloadingKey) return;

    setDownloadingKey(resourceKey);
    setError("");
    setMessage("");

    try {
      const response = await api.get(
        `/courses/lectures/${lectureId}/resources/download`,
        {
          params: { key: resourceKey },
        },
      );

      const downloadUrl = getResponseData<{ url: string }>(response)?.url;

      if (typeof downloadUrl !== "string" || !downloadUrl.trim()) {
        throw new Error("Download URL was not returned.");
      }

      const link = document.createElement("a");
      link.href = downloadUrl;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
      link.click();

      setMessage("Your resource download link is ready.");
    } catch (err) {
      console.error("Resource download failed:", err);
      setError("Could not download this resource. Please try again.");
    } finally {
      setDownloadingKey(null);
    }
  }

  function formatTime(seconds: number) {
    const safeSeconds = Math.max(0, Math.floor(seconds));
    const hours = Math.floor(safeSeconds / 3600);
    const minutes = Math.floor((safeSeconds % 3600) / 60);
    const remainingSeconds = safeSeconds % 60;

    if (hours > 0) {
      return `${hours}:${String(minutes).padStart(2, "0")}:${String(
        remainingSeconds,
      ).padStart(2, "0")}`;
    }

    return `${minutes}:${String(remainingSeconds).padStart(2, "0")}`;
  }

  function seekTo(seconds: number) {
    const video = videoRef.current;

    if (video) {
      video.currentTime = seconds;
    }
  }

  if (loading) {
    return <main className="p-8">Loading course...</main>;
  }

  if (!course) {
    return (
      <main className="p-8">
        <p role="alert" className="mb-4 text-destructive">
          {error || "Course not found."}
        </p>

        <Link to="/my-courses" className="underline">
          Back to My Courses
        </Link>
      </main>
    );
  }

  const modules = course.modules ?? [];

  return (
    <main className="min-h-screen bg-background px-4 py-8 text-foreground md:px-8">
      <div className="mx-auto max-w-7xl">
        <Link to="/my-courses" className="text-sm underline">
          ← My Courses
        </Link>

        <h1 className="mb-2 mt-4 text-3xl font-bold">{course.title}</h1>

        {course.description && (
          <p className="mb-6 text-muted-foreground">{course.description}</p>
        )}

        {message && (
          <p role="status" className="mb-4 text-sm">
            {message}
          </p>
        )}

        {error && (
          <p role="alert" className="mb-4 text-sm text-destructive">
            {error}
          </p>
        )}

        <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_320px]">
          <section className="min-w-0">
            {activeLecture ? (
              <>
                <div className="overflow-hidden rounded-xl bg-black">
                  {activeLecture.video_url ? (
                    <video
                      ref={videoRef}
                      key={activeLecture.id}
                      id="course-lecture-video"
                      className="aspect-video w-full"
                      controls
                      playsInline
                      preload="metadata"
                      src={activeLecture.video_url}
                      onLoadedMetadata={(event) => {
                        const video = event.currentTarget;
                        const savedSeconds = progress?.watched_seconds ?? 0;

                        if (
                          savedSeconds > 0 &&
                          Number.isFinite(video.duration) &&
                          savedSeconds < video.duration
                        ) {
                          video.currentTime = savedSeconds;
                        }
                      }}
                      onTimeUpdate={(event) => {
                        const video = event.currentTarget;
                        const currentSecond = Math.floor(video.currentTime);

                        if (
                          currentSecond > 0 &&
                          currentSecond % 15 === 0 &&
                          currentSecond !== lastSavedSecondRef.current
                        ) {
                          lastSavedSecondRef.current = currentSecond;
                          void saveProgress(video.currentTime, false);
                        }
                      }}
                      onEnded={(event) => {
                        void saveProgress(event.currentTarget.duration, true);
                      }}
                    />
                  ) : (
                    <p className="p-8 text-white">No video available.</p>
                  )}
                </div>

                <h2 className="mt-4 text-2xl font-semibold">
                  {activeLecture.title}
                </h2>

                {progress?.is_completed && (
                  <p className="mt-2 text-sm text-green-600">
                    ✓ Lecture completed
                  </p>
                )}

                <div className="mt-4 flex flex-wrap gap-3">
                  <button
                    type="button"
                    onClick={() => {
                      if (videoRef.current) {
                        void saveProgress(videoRef.current.currentTime, true);
                      }
                    }}
                    className="rounded-lg border px-4 py-2"
                  >
                    Mark as completed
                  </button>

                  <button
                    type="button"
                    onClick={() => void addBookmark()}
                    className="rounded-lg border px-4 py-2"
                  >
                    Bookmark current time
                  </button>
                </div>

                {activeLecture.resource_urls &&
                  activeLecture.resource_urls.length > 0 && (
                    <section className="mt-6 rounded-xl border p-5">
                      <h3 className="text-lg font-semibold">
                        Lecture Resources
                      </h3>

                      <p className="mt-1 text-sm text-muted-foreground">
                        {activeLecture.resource_urls.length} resource file
                        {activeLecture.resource_urls.length === 1
                          ? ""
                          : "s"}{" "}
                        available.
                      </p>

                      <ul className="mt-3 space-y-3">
                        {activeLecture.resource_urls.map((resourceKey) => {
                          const fileName =
                            resourceKey.split("/").pop() ?? "Resource file";

                          return (
                            <li
                              key={resourceKey}
                              className="flex flex-wrap items-center justify-between gap-3 rounded-lg bg-muted p-3"
                            >
                              <div className="min-w-0">
                                <p className="break-all text-sm font-medium">
                                  {fileName}
                                </p>

                                <p className="text-xs text-muted-foreground">
                                  Lecture PDF / Slides
                                </p>
                              </div>

                              <button
                                type="button"
                                disabled={downloadingKey !== null}
                                onClick={() =>
                                  void downloadResource(
                                    activeLecture.id,
                                    resourceKey,
                                  )
                                }
                                className="rounded-lg bg-primary px-4 py-2 text-sm text-primary-foreground disabled:opacity-50"
                              >
                                {downloadingKey === resourceKey
                                  ? "Preparing..."
                                  : "Download"}
                              </button>
                            </li>
                          );
                        })}
                      </ul>
                    </section>
                  )}

                <section className="mt-8 rounded-xl border p-5">
                  <h3 className="text-lg font-semibold">Lecture Notes</h3>

                  <textarea
                    value={noteContent}
                    onChange={(event) => setNoteContent(event.target.value)}
                    placeholder="Write a note about this moment in the lecture..."
                    rows={4}
                    className="mt-3 w-full rounded-lg border bg-background p-3"
                  />

                  <button
                    type="button"
                    disabled={saving || !noteContent.trim()}
                    onClick={() => void addNote()}
                    className="mt-3 rounded-lg bg-primary px-4 py-2 text-primary-foreground disabled:opacity-50"
                  >
                    {saving ? "Saving..." : "Save note at current time"}
                  </button>

                  <ul className="mt-5 space-y-3">
                    {notes.map((note) => (
                      <li key={note.id} className="rounded-lg bg-muted p-3">
                        <button
                          type="button"
                          className="font-medium underline"
                          onClick={() => seekTo(note.timestamp_seconds)}
                        >
                          {formatTime(note.timestamp_seconds)}
                        </button>

                        <p className="mt-1 whitespace-pre-wrap">
                          {note.content}
                        </p>
                      </li>
                    ))}

                    {notes.length === 0 && (
                      <li className="text-sm text-muted-foreground">
                        No notes for this lecture yet.
                      </li>
                    )}
                  </ul>
                </section>

                <section className="mt-6 rounded-xl border p-5">
                  <h3 className="text-lg font-semibold">Bookmarks</h3>

                  <ul className="mt-3 flex flex-wrap gap-2">
                    {bookmarks.map((bookmark) => (
                      <li key={bookmark.id}>
                        <button
                          type="button"
                          className="rounded-lg border px-3 py-2 underline"
                          onClick={() => seekTo(bookmark.timestamp_seconds)}
                        >
                          {formatTime(bookmark.timestamp_seconds)}
                        </button>
                      </li>
                    ))}

                    {bookmarks.length === 0 && (
                      <li className="text-sm text-muted-foreground">
                        No bookmarks yet.
                      </li>
                    )}
                  </ul>
                </section>
              </>
            ) : (
              <p className="rounded-xl border p-8">
                {message || "No lecture is available for this course yet."}
              </p>
            )}
          </section>

          <aside className="rounded-xl border p-4">
            <h2 className="text-lg font-semibold">Course Content</h2>

            <div className="mt-4 space-y-5">
              {modules.map((module) => (
                <section key={module.id}>
                  <h3 className="mb-2 font-medium">{module.title}</h3>

                  <ul className="space-y-2">
                    {module.lectures.map((lecture) => (
                      <li key={lecture.id}>
                        <button
                          type="button"
                          onClick={() => void loadLectureData(lecture)}
                          className={`w-full rounded-lg border p-3 text-left text-sm ${
                            activeLecture?.id === lecture.id
                              ? "border-primary bg-muted"
                              : ""
                          }`}
                        >
                          <span>{lecture.title}</span>

                          {activeLecture?.id === lecture.id &&
                            progress?.is_completed && (
                              <span className="ml-2 text-green-600">✓</span>
                            )}
                        </button>
                      </li>
                    ))}
                  </ul>
                </section>
              ))}

              {modules.length === 0 && (
                <p className="text-sm text-muted-foreground">
                  No course modules available.
                </p>
              )}
            </div>
          </aside>
        </div>
      </div>
    </main>
  );
}


import { useEffect, useState } from "react";
import { useDispatch, useSelector } from "react-redux";
import { Link, useNavigate } from "react-router";

import api from "@/services/api";
import { clearTokens, getAccessToken } from "@/services/tokenStorage";
import { clearAuth, setUser } from "@/store/authSlice";
import type { AppDispatch, RootState } from "@/store";

type InstructorCourse = {
  id: string;
  title: string;
  status: "pending" | "approved" | "rejected" | string;
  created_at?: string;
};

export default function DashboardPage() {
  const user = useSelector((state: RootState) => state.auth.user);
  const dispatch = useDispatch<AppDispatch>();
  const navigate = useNavigate();

  const [loggingOut, setLoggingOut] = useState(false);
  const [error, setError] = useState("");

  const [instructorCourses, setInstructorCourses] = useState<
    InstructorCourse[]
  >([]);
  const [coursesLoading, setCoursesLoading] = useState(false);
  const [coursesError, setCoursesError] = useState("");

  const isInstructor = user?.roles?.includes("instructor") ?? false;
  const isAdmin = user?.roles?.includes("admin") ?? false;
  const isStudent = user?.roles?.includes("student") ?? false;

  useEffect(() => {
    let cancelled = false;

    async function loadProfile() {
      if (!getAccessToken()) {
        return;
      }

      try {
        const response = await api.get("/auth/me");

        if (!cancelled && response.data?.success && response.data?.data) {
          dispatch(setUser(response.data.data));
        }
      } catch (error) {
        console.error("Failed to restore user profile:", error);
      }
    }

    void loadProfile();

    return () => {
      cancelled = true;
    };
  }, [dispatch]);

  useEffect(() => {
    if (!isInstructor || !getAccessToken()) {
      setInstructorCourses([]);
      setCoursesLoading(false);
      setCoursesError("");
      return;
    }

    let cancelled = false;

    async function loadInstructorCourses() {
      setCoursesLoading(true);
      setCoursesError("");

      try {
        const response = await api.get("/courses/instructor/me");

        if (!cancelled && response.data?.success) {
          setInstructorCourses(response.data.data ?? []);
        }
      } catch (error) {
        console.error("Failed to load instructor courses:", error);

        if (!cancelled) {
          setCoursesError(
            "Failed to load your courses. Please refresh the page.",
          );
        }
      } finally {
        if (!cancelled) {
          setCoursesLoading(false);
        }
      }
    }

    void loadInstructorCourses();

    return () => {
      cancelled = true;
    };
  }, [isInstructor]);

  async function handleLogout() {
    if (loggingOut) {
      return;
    }

    setLoggingOut(true);
    setError("");

    try {
      await api.post("/auth/logout");
    } catch (error) {
      console.error("Logout API request failed:", error);
      setError(
        "The logout request failed. Your local session will still be cleared.",
      );
    } finally {
      clearTokens();
      dispatch(clearAuth());
      navigate("/login", { replace: true });
      setLoggingOut(false);
    }
  }

  return (
    <main className="min-h-screen bg-background px-6 py-10 text-foreground">
      <div className="mx-auto max-w-5xl">
        <div className="flex flex-wrap items-center justify-between gap-4">
          <h1 className="text-3xl font-bold">Dashboard</h1>

          <button
            type="button"
            onClick={handleLogout}
            disabled={loggingOut}
            className="rounded-md border px-4 py-2 text-sm disabled:opacity-50"
          >
            {loggingOut ? "Signing out..." : "Logout"}
          </button>
        </div>

        {error && (
          <p role="alert" className="mt-3 text-sm text-destructive">
            {error}
          </p>
        )}

        <p className="mt-4 text-muted-foreground">
          Welcome back, {user?.full_name ?? "User"}!
        </p>

        <section className="mt-8 rounded-xl border bg-card p-6">
          <h2 className="text-xl font-semibold">Your Account</h2>

          <p className="mt-3 text-sm">
            <span className="font-medium">Name:</span>{" "}
            {user?.full_name ?? "Not available"}
          </p>

          <p className="mt-2 text-sm">
            <span className="font-medium">Email:</span>{" "}
            {user?.email ?? "Not available"}
          </p>

          <p className="mt-2 text-sm">
            <span className="font-medium">Role:</span>{" "}
            {user?.roles?.join(", ") || "Not available"}
          </p>
        </section>

        <section className="mt-6 flex flex-wrap gap-3">
          {isAdmin && (
            <Link
              to="/admin/courses"
              className="inline-flex items-center rounded-lg bg-slate-900 px-4 py-2 text-sm font-medium text-white hover:bg-slate-800"
            >
              Review Pending Courses
            </Link>
          )}

          {isInstructor && (
            <>
              <Link
                to="/instructor/courses/new"
                className="inline-flex rounded-md bg-primary px-4 py-2 text-primary-foreground"
              >
                Create Course
              </Link>

              <Link
                to="/instructor/courses"
                className="inline-flex items-center rounded-md border px-4 py-2 text-sm hover:bg-muted"
              >
                My Courses &amp; Approval Status
              </Link>
            </>
          )}

          <Link
            to="/courses"
            className="inline-flex items-center rounded-md border px-4 py-2 hover:bg-muted"
          >
            Browse Courses
          </Link>

          {isStudent && (
            <Link
              to="/my-courses"
              className="inline-flex items-center rounded-md border px-4 py-2 hover:bg-muted"
            >
              My Courses
            </Link>
          )}
        </section>

        {isInstructor && (
          <section className="mt-8 rounded-xl border bg-card p-6">
            <h2 className="text-xl font-semibold">My Courses</h2>

            {coursesLoading && (
              <p className="mt-3 text-sm text-muted-foreground">
                Loading your courses...
              </p>
            )}

            {coursesError && (
              <p role="alert" className="mt-3 text-sm text-destructive">
                {coursesError}
              </p>
            )}

            {!coursesLoading &&
              !coursesError &&
              instructorCourses.length === 0 && (
                <p className="mt-3 text-sm text-muted-foreground">
                  You haven't created any courses yet.
                </p>
              )}

            {!coursesLoading && instructorCourses.length > 0 && (
              <div className="mt-4 space-y-4">
                {instructorCourses.map((course) => (
                  <div key={course.id} className="rounded-lg border p-4">
                    <h3 className="font-medium">{course.title}</h3>

                    <p className="mt-2 text-sm">
                      Status:{" "}
                      <span className="font-semibold capitalize">
                        {course.status}
                      </span>
                    </p>

                    {course.status === "pending" && (
                      <p className="mt-2 text-sm text-muted-foreground">
                        Your course is awaiting admin approval. You can manage
                        its content after approval.
                      </p>
                    )}

                    {course.status === "rejected" && (
                      <p className="mt-2 text-sm text-destructive">
                        Your course was rejected. Please contact the admin for
                        further instructions.
                      </p>
                    )}

                    {course.status === "approved" && (
                      <Link
                        to={`/instructor/courses/${course.id}/content`}
                        className="mt-3 inline-flex rounded-md bg-primary px-4 py-2 text-sm text-primary-foreground"
                      >
                        Manage Content
                      </Link>
                    )}
                  </div>
                ))}
              </div>
            )}
          </section>
        )}
      </div>
    </main>
  );
}


import { useEffect, useState } from "react";
import { Link } from "react-router";
import api from "@/services/api";

interface Enrollment {
  enrollment_id: string;
  course_id: string;
  enrolled_at: string;
  progress_percentage: number | string;

function getErrorMessage(error: unknown, fallback: string): string {
  if (axios.isAxiosError(error)) {

  }

  if (error instanceof Error) {
    return error.message;
  }

  return fallback;
}

function getNextOrder(items: { order_index: number }[]): number {
  return Math.max(0, ...items.map((item) => item.order_index)) + 1;
}

          .map((module) => ({
            ...module,
            lectures: [...(module.lectures ?? [])].sort(
              (a, b) => a.order_index - b.order_index,
            ),
          }))
          .sort((a, b) => a.order_index - b.order_index);

        setModules(existingModules);
        setSelectedModuleId(existingModules[0]?.id ?? "");

          setError(

        if (!cancelled) {
          setLoadingContent(false);

        setSelectedModuleId(nextModule?.id ?? "");
        setLectureOrder(String(getNextOrder(nextModule?.lectures ?? [])));
      }

      if (editingModuleId === module.id) {
        cancelEditingModule();
      }

      setMessage("Module deleted successfully.");
    } catch (err) {
      setError(getErrorMessage(err, "Failed to delete module."));
    } finally {
      setLoading(false);
    }
  }

  function handleVideoChange(event: ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0] ?? null;

    clearNotifications();

    if (!file) {
      setVideoFile(null);
      return;
    }

    if (!file.type.startsWith("video/")) {
      setVideoFile(null);
      event.target.value = "";
      setError("Please select a valid video file.");
      return;
    }

    setVideoFile(file);
  }

  async function handleUploadLecture(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (loading) return;

    if (!selectedModuleId) {
      setError("Please select a module.");
      setMessage("");
      return;
    }

    if (!videoFile) {
      setError("Please select a video file.");
      setMessage("");
      return;
    }

    const title = lectureTitle.trim();
    const order = Number(lectureOrder);

    if (!title || !Number.isInteger(order) || order < 1) {
      setError("Enter a valid lecture title and order.");
      setMessage("");
      return;
    }

    setLoading(true);
    clearNotifications();

    try {
      const formData = new FormData();

      formData.append("title", title);
      formData.append("order_index", String(order));
      formData.append("video", videoFile);

      const response = await api.post<ApiResponse<Lecture>>(
        `/courses/modules/${selectedModuleId}/video`,
        formData,
      );

      const createdLecture = response.data?.data;

      if (!response.data?.success || !createdLecture?.id) {
        throw new Error("Lecture video upload failed.");
      }

      setModules((current) =>
        current.map((module) =>
          module.id === selectedModuleId
            ? {
                ...module,
                lectures: [
                  ...(module.lectures ?? []).filter(
                    (lecture) => lecture.id !== createdLecture.id,
                  ),
                  createdLecture,
                ].sort((a, b) => a.order_index - b.order_index),
              }
            : module,
        ),
      );

      setLectureTitle("");
      setLectureOrder(String(order + 1));
      setVideoFile(null);

      const fileInput = document.getElementById(
        "lecture-video",
      ) as HTMLInputElement | null;

      if (fileInput) {
        fileInput.value = "";
      }

      setMessage("Lecture created and video uploaded successfully.");
    } catch (err) {
      setError(getErrorMessage(err, "Failed to upload lecture video."));
    } finally {
      setLoading(false);
    }
  }

  function handleResourceFileChange(event: ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0] ?? null;

    clearNotifications();

    if (!file) {
      setResourceFile(null);
      return;
    }

    const allowedExtensions = [".pdf", ".ppt", ".pptx"];
    const extension = file.name.slice(file.name.lastIndexOf(".")).toLowerCase();

    if (!allowedExtensions.includes(extension)) {
      setResourceFile(null);
      event.target.value = "";
      setError("Only PDF, PPT, and PPTX files are allowed.");
      return;
    }

    if (file.size > 20 * 1024 * 1024) {
      setResourceFile(null);
      event.target.value = "";
      setError("Resource files must be 20 MB or smaller.");
      return;
    }

    setResourceFile(file);
  }

  async function handleUploadResource(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (loading) return;

    if (!selectedLectureId || !resourceFile) {
      setError("Select a lecture and a resource file.");
      setMessage("");
      return;
    }

    setLoading(true);
    clearNotifications();

    try {
      const formData = new FormData();
      formData.append("file", resourceFile);

      const response = await api.post<ApiResponse<Lecture>>(
        `/courses/lectures/${selectedLectureId}/resources`,
        formData,
      );

      if (!response.data?.success) {
        throw new Error("Resource upload failed.");
      }

      const updatedLecture = response.data.data;

      setModules((current) =>
        current.map((module) => ({
          ...module,
          lectures: (module.lectures ?? []).map((lecture) =>
            lecture.id === selectedLectureId
              ? {
                  ...lecture,
                  resource_urls: updatedLecture?.resource_urls ?? [
                    ...(lecture.resource_urls ?? []),
                  ],
                }
              : lecture,
          ),
        })),
      );

      setResourceFile(null);
      setSelectedLectureId("");
      setMessage("Lecture resource uploaded successfully.");

      const input = document.getElementById(
        "lecture-resource",
      ) as HTMLInputElement | null;

      if (input) {
        input.value = "";
      }
    } catch (err) {
      setError(getErrorMessage(err, "Failed to upload lecture resource."));
    } finally {
      setLoading(false);
    }
  }

  if (!courseId) {
    return (
      <main className="p-6">
        <p role="alert">Course ID is missing.</p>

        <button
          type="button"
          onClick={() => navigate("/instructor/courses/new")}
          className="mt-4 rounded-lg border px-4 py-2"
        >
          Back to Create Course
        </button>
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-background px-4 py-8 text-foreground sm:px-6 sm:py-10">
      <div className="mx-auto max-w-3xl space-y-8">
        <button
          type="button"
          onClick={() => navigate("/instructor/courses/new")}
          className="text-sm text-muted-foreground hover:underline"
        >
          ← Back to Create Course
        </button>

        <header>
          <h1 className="text-3xl font-bold">Course Content Manager</h1>

          <p className="mt-2 text-sm text-muted-foreground">
            Create, edit, and delete modules, and upload lecture videos to
            MinIO.
          </p>

          <p className="mt-1 break-all text-xs text-muted-foreground">
            Course ID: {courseId}
          </p>
        </header>

        {error && (
          <p
            role="alert"
            className="rounded-lg border border-destructive p-3 text-sm"
          >
            {error}
          </p>
        )}

        {message && (
          <p role="status" className="rounded-lg border p-3 text-sm">
            {message}
          </p>
        )}

        {loadingContent ? (
          <section className="rounded-xl border bg-card p-6" aria-live="polite">
            Loading course content...
          </section>
        ) : (
          <>
            <section className="space-y-4 rounded-xl border bg-card p-6">
              <h2 className="text-xl font-semibold">1. Create Module</h2>

              <form onSubmit={handleCreateModule} className="space-y-4">
                <div>
                  <label
                    htmlFor="module-title"
                    className="mb-2 block font-medium"
                  >
                    Module title
                  </label>

                  <input
                    id="module-title"
                    type="text"
                    required
                    maxLength={200}
                    value={moduleTitle}
                    onChange={(event) => setModuleTitle(event.target.value)}
                    placeholder="Introduction"
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />
                </div>

                <div>
                  <label
                    htmlFor="module-order"
                    className="mb-2 block font-medium"
                  >
                    Module order
                  </label>

                  <input
                    id="module-order"
                    type="number"
                    min="1"
                    step="1"
                    required
                    value={moduleOrder}
                    onChange={(event) => setModuleOrder(event.target.value)}
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />
                </div>

                <button
                  type="submit"
                  disabled={loading}
                  className="rounded-lg bg-primary px-4 py-2 text-primary-foreground disabled:opacity-50"
                >
                  {loading ? "Please wait..." : "Create Module"}
                </button>
              </form>
            </section>

            <section className="space-y-4 rounded-xl border bg-card p-6">
              <h2 className="text-xl font-semibold">2. Manage Modules</h2>

              {modules.length === 0 ? (
                <p className="text-sm text-muted-foreground">
                  No modules found. Create your first module above.
                </p>
              ) : (
                <div className="space-y-3">
                  {modules.map((module) => (
                    <div
                      key={module.id}
                      className="flex flex-wrap items-center justify-between gap-3 rounded-lg border p-4"
                    >
                      <div className="min-w-0">
                        <p className="break-words font-medium">
                          {module.order_index}. {module.title}
                        </p>

                        <p className="text-sm text-muted-foreground">
                          {module.lectures?.length ?? 0} lectures
                        </p>
                      </div>

                      <div className="flex flex-wrap gap-2">
                        <button
                          type="button"
                          onClick={() => startEditingModule(module)}
                          disabled={loading}
                          className="rounded-md border px-3 py-2 text-sm hover:bg-muted disabled:opacity-50"
                        >
                          Edit
                        </button>

                        <button
                          type="button"
                          onClick={() => void handleDeleteModule(module)}
                          disabled={loading}
                          className="rounded-md border border-destructive px-3 py-2 text-sm text-destructive hover:bg-destructive/10 disabled:opacity-50"
                        >
                          Delete
                        </button>
                      </div>
                    </div>
                  ))}
                </div>
              )}

              {editingModuleId && (
                <form
                  onSubmit={handleUpdateModule}
                  className="space-y-4 border-t pt-4"
                >
                  <h3 className="font-semibold">Edit Module</h3>

                  <div>
                    <label
                      htmlFor="edit-module-title"
                      className="mb-2 block text-sm font-medium"
                    >
                      Module title
                    </label>

                    <input
                      id="edit-module-title"
                      type="text"
                      required
                      maxLength={200}
                      value={editingModuleTitle}
                      onChange={(event) =>
                        setEditingModuleTitle(event.target.value)
                      }
                      className="w-full rounded-lg border bg-background px-3 py-2"
                    />
                  </div>

                  <div>
                    <label
                      htmlFor="edit-module-order"
                      className="mb-2 block text-sm font-medium"
                    >
                      Module order
                    </label>

                    <input
                      id="edit-module-order"
                      type="number"
                      min="1"
                      step="1"
                      required
                      value={editingModuleOrder}
                      onChange={(event) =>
                        setEditingModuleOrder(event.target.value)
                      }
                      className="w-full rounded-lg border bg-background px-3 py-2"
                    />
                  </div>

                  <div className="flex flex-wrap gap-3">
                    <button
                      type="submit"
                      disabled={loading}
                      className="rounded-lg bg-primary px-4 py-2 text-primary-foreground disabled:opacity-50"
                    >
                      {loading ? "Saving..." : "Save Changes"}
                    </button>

                    <button
                      type="button"
                      disabled={loading}
                      onClick={cancelEditingModule}
                      className="rounded-lg border px-4 py-2 disabled:opacity-50"
                    >
                      Cancel
                    </button>
                  </div>
                </form>
              )}
            </section>

            <section className="space-y-4 rounded-xl border bg-card p-6">
              <h2 className="text-xl font-semibold">3. Upload Lecture Video</h2>

              <div>
                <label
                  htmlFor="module-select"
                  className="mb-2 block font-medium"
                >
                  Select a module
                </label>

                <select
                  id="module-select"
                  required
                  value={selectedModuleId}
                  onChange={(event) =>
                    handleModuleSelection(event.target.value)
                  }
                  className="w-full rounded-lg border bg-background px-3 py-2"
                >
                  <option value="">Choose a module</option>

                  {modules.map((module) => (
                    <option key={module.id} value={module.id}>
                      {module.order_index}. {module.title}
                    </option>
                  ))}
                </select>
              </div>

              <form onSubmit={handleUploadLecture} className="space-y-4">
                <div>
                  <label
                    htmlFor="lecture-title"
                    className="mb-2 block font-medium"
                  >
                    Lecture title
                  </label>

                  <input
                    id="lecture-title"
                    type="text"
                    required
                    maxLength={200}
                    value={lectureTitle}
                    onChange={(event) => setLectureTitle(event.target.value)}
                    placeholder="Getting started"
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />
                </div>

                <div>
                  <label
                    htmlFor="lecture-order"
                    className="mb-2 block font-medium"
                  >
                    Lecture order
                  </label>

                  <input
                    id="lecture-order"
                    type="number"
                    min="1"
                    step="1"
                    required
                    value={lectureOrder}
                    onChange={(event) => setLectureOrder(event.target.value)}
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />
                </div>

                <div>
                  <label
                    htmlFor="lecture-video"
                    className="mb-2 block font-medium"
                  >
                    Video file
                  </label>

                  <input
                    id="lecture-video"
                    type="file"
                    accept="video/*"
                    required
                    onChange={handleVideoChange}
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />

                  {videoFile && (
                    <p className="mt-2 break-all text-sm text-muted-foreground">
                      Selected: {videoFile.name} (
                      {(videoFile.size / (1024 * 1024)).toFixed(1)} MB)
                    </p>
                  )}
                </div>

                <button
                  type="submit"
                  disabled={
                    loading ||
                    !selectedModuleId ||
                    !videoFile ||
                    modules.length === 0
                  }
                  className="w-full rounded-lg bg-primary px-4 py-3 font-medium text-primary-foreground disabled:opacity-50"
                >
                  {loading ? "Uploading..." : "Create Lecture & Upload Video"}
                </button>
              </form>
            </section>

            <section className="space-y-4 rounded-xl border bg-card p-6">
              <h2 className="text-xl font-semibold">4. Upload PDF / Slides</h2>

              <form onSubmit={handleUploadResource} className="space-y-4">
                <div>
                  <label
                    htmlFor="resource-lecture"
                    className="mb-2 block font-medium"
                  >
                    Select lecture
                  </label>

                  <select
                    id="resource-lecture"
                    required
                    value={selectedLectureId}
                    onChange={(event) =>
                      setSelectedLectureId(event.target.value)
                    }
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  >
                    <option value="">Choose a lecture</option>

                    {lectures.map((lecture) => (
                      <option key={lecture.id} value={lecture.id}>
                        {lecture.moduleTitle} — {lecture.title}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label
                    htmlFor="lecture-resource"
                    className="mb-2 block font-medium"
                  >
                    PDF or PowerPoint file
                  </label>

                  <input
                    id="lecture-resource"
                    type="file"
                    accept=".pdf,.ppt,.pptx"
                    required
                    onChange={handleResourceFileChange}
                    className="w-full rounded-lg border bg-background px-3 py-2"
                  />

                  {resourceFile && (
                    <p className="mt-2 break-all text-sm text-muted-foreground">
                      Selected: {resourceFile.name} (
                      {(resourceFile.size / (1024 * 1024)).toFixed(1)} MB)
                    </p>
                  )}
                </div>

                <button
                  type="submit"
                  disabled={loading || !selectedLectureId || !resourceFile}
                  className="w-full rounded-lg bg-primary px-4 py-3 font-medium text-primary-foreground disabled:opacity-50"
                >
                  {loading ? "Uploading..." : "Upload Resource"}
                </button>
              </form>
            </section>

            <section className="rounded-xl border bg-card p-6">
              <h2 className="text-xl font-semibold">5. Uploaded Lectures</h2>

              {lectures.length === 0 ? (
                <p className="mt-3 text-sm text-muted-foreground">
                  No lectures have been uploaded yet.
                </p>
              ) : (
                <div className="mt-4 space-y-4">
                  {modules
                    .filter((module) => (module.lectures?.length ?? 0) > 0)
                    .map((module) => (
                      <div key={module.id} className="space-y-2">
                        <h3 className="font-medium">
                          {module.order_index}. {module.title}
                        </h3>

                        <ul className="space-y-2">
                          {[...(module.lectures ?? [])]
                            .sort((a, b) => a.order_index - b.order_index)
                            .map((lecture) => (
                              <li
                                key={lecture.id}
                                className="rounded-lg border p-3"
                              >
                                <p className="font-medium">
                                  {lecture.order_index}. {lecture.title}
                                </p>

                                <p className="mt-1 text-xs text-muted-foreground">
                                  {lecture.video_url
                                    ? "Video uploaded"
                                    : "Lecture created"}
                                </p>

                                <p className="text-sm text-muted-foreground">
                                  {(lecture.resource_urls?.length ?? 0) > 0
                                    ? `${lecture.resource_urls!.length} resource file(s) uploaded`
                                    : "No PDF or slides uploaded"}
                                </p>
                              </li>
                            ))}
                        </ul>
                      </div>
                    ))}
                </div>
              )}
            </section>
          </>
        )}
      </div>
    </main>
  );
}


import { useState, type FormEvent, type ChangeEvent } from "react";
import { Link, useNavigate } from "react-router";
import { useSelector } from "react-redux";
import axios from "axios";
import api from "@/services/api";
import type { RootState } from "@/store";

type CreatedCourse = {
  id?: string;
  course_id?: string;
  title?: string;
};

type ApiResponse<T> = {
  success: boolean;
  data?: T;
  message?: string;
};

function getErrorMessage(error: unknown, fallback: string): string {
  if (axios.isAxiosError(error)) {
    return error.response?.data?.message || fallback;
  }

  if (error instanceof Error) {
    return error.message;
  }

  return fallback;
}

export default function CreateCoursePage() {
  const navigate = useNavigate();

  const user = useSelector((state: RootState) => state.auth.user);

  const [thumbnail, setThumbnail] = useState<File | null>(null);
  const [title, setTitle] = useState("");
  const [description, setDescription] = useState("");
  const [category, setCategory] = useState("");
  const [difficulty, setDifficulty] = useState("beginner");
  const [price, setPrice] = useState("0");

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  const isInstructor = user?.roles?.includes("instructor") ?? false;

  function handleThumbnailChange(event: ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0] ?? null;

    setError("");
    setSuccess("");

    if (!file) {
      setThumbnail(null);
      return;
    }

    const allowedTypes = ["image/jpeg", "image/png", "image/webp"];

    if (!allowedTypes.includes(file.type)) {
      setThumbnail(null);
      event.target.value = "";
      setError("Please select a JPEG, PNG, or WebP image.");
      return;
    }

    if (file.size > 5 * 1024 * 1024) {
      setThumbnail(null);
      event.target.value = "";
      setError("Thumbnail size must not exceed 5 MB.");
      return;
    }

    setThumbnail(file);
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (!isInstructor || loading) {
      return;
    }

    setError("");
    setSuccess("");

    const trimmedTitle = title.trim();
    const trimmedDescription = description.trim();
    const trimmedCategory = category.trim();
    const parsedPrice = Number(price);

    if (!trimmedTitle) {
      setError("Course title is required.");
      return;
    }

    if (
      !Number.isFinite(parsedPrice) ||
      parsedPrice < 0 ||
      !Number.isInteger(parsedPrice * 100)
    ) {
      setError("Enter a valid non-negative course price.");
      return;
    }

    if (!["beginner", "intermediate", "advanced"].includes(difficulty)) {
      setError("Please select a valid difficulty level.");
      return;
    }

    setLoading(true);

    try {
      const formData = new FormData();

      formData.append("title", trimmedTitle);
      formData.append("description", trimmedDescription);
      formData.append("category", trimmedCategory);
      formData.append("difficulty", difficulty);
      formData.append("price", String(parsedPrice));

      if (thumbnail) {
        formData.append("thumbnail", thumbnail);
      }

      const response = await api.post<ApiResponse<CreatedCourse>>(
        "/courses",
        formData,
      );

      if (!response.data?.success) {
        throw new Error(response.data?.message || "Course creation failed.");
      }

      const createdCourse = response.data.data;

      const courseId = createdCourse?.id ?? createdCourse?.course_id;

      if (!courseId) {
        throw new Error(
          "Course creation response did not include a course ID.",
        );
      }

      setSuccess("Course created successfully. Redirecting...");

      navigate(`/instructor/courses/${courseId}/content`);
    } catch (err) {
      setError(getErrorMessage(err, "Failed to create course."));
    } finally {
      setLoading(false);
    }
  }

  if (!isInstructor) {
    return (
      <main className="mx-auto max-w-3xl px-6 py-12">
        <h1 className="text-2xl font-bold">Instructor Access Required</h1>

        <p className="mt-3 text-muted-foreground">
          Only instructors can create courses.
        </p>

        <Link to="/dashboard" className="mt-4 inline-block underline">
          Back to Dashboard
        </Link>
      </main>
    );
  }

  return (
    <main className="min-h-screen bg-background px-4 py-10 text-foreground sm:px-6">
      <div className="mx-auto max-w-3xl">
        <button
          type="button"
          onClick={() => navigate("/dashboard")}
          className="mb-6 text-sm text-muted-foreground hover:underline"
        >
          ← Back to Dashboard
        </button>

        <h1 className="text-3xl font-bold">Create Course</h1>

        <p className="mt-2 text-muted-foreground">
          Submit a new course for admin approval.
        </p>

        <form
          onSubmit={handleSubmit}
          className="mt-8 space-y-5 rounded-xl border bg-card p-6"
        >
          <div>
            <label htmlFor="title" className="mb-2 block font-medium">
              Course Title *
            </label>

            <input
              id="title"
              type="text"
              required
              maxLength={200}
              value={title}
              onChange={(event) => setTitle(event.target.value)}
              placeholder="Introduction to React"
              className="w-full rounded-lg border bg-background px-3 py-2"
            />
          </div>

          <div>
            <label htmlFor="description" className="mb-2 block font-medium">
              Description
            </label>

            <textarea
              id="description"
              rows={4}
              value={description}
              onChange={(event) => setDescription(event.target.value)}
              placeholder="What will students learn?"
              className="w-full rounded-lg border bg-background px-3 py-2"
            />
          </div>

          <div>
            <label htmlFor="thumbnail" className="mb-2 block font-medium">
              Course Thumbnail
            </label>

            <input
              id="thumbnail"
              name="thumbnail"
              type="file"
              accept="image/jpeg,image/png,image/webp"
              onChange={handleThumbnailChange}
              className="block w-full"
            />

            <p className="mt-1 text-sm text-muted-foreground">
              JPEG, PNG, or WebP. Maximum size: 5 MB.
            </p>

            {thumbnail && (
              <p className="mt-2 break-all text-sm text-muted-foreground">
                Selected: {thumbnail.name} (
                {(thumbnail.size / (1024 * 1024)).toFixed(2)} MB)
              </p>
            )}
          </div>

          <div>
            <label htmlFor="category" className="mb-2 block font-medium">
              Category
            </label>

            <input
              id="category"
              type="text"
              maxLength={100}
              value={category}
              onChange={(event) => setCategory(event.target.value)}
              placeholder="Programming"
              className="w-full rounded-lg border bg-background px-3 py-2"
            />
          </div>

          <div>
            <label htmlFor="difficulty" className="mb-2 block font-medium">
              Difficulty
            </label>

            <select
              id="difficulty"
              value={difficulty}
              onChange={(event) => setDifficulty(event.target.value)}
              className="w-full rounded-lg border bg-background px-3 py-2"
            >

import { useCallback, useEffect, useState, type FormEvent } from "react";
import { Link, useParams } from "react-router";
import axios from "axios";
import api from "@/services/api";

type Module = {
  id: string;
  title: string;
  order_index: number;
};

type Quiz = {
  id: string;
  title: string;
  module_id: string;
  is_ai_generated: boolean;
  approval_status: "pending" | "approved" | "rejected";
  generated_from_lecture_id: string | null;
};

type ApiResponse<T> = {
  success: boolean;
  data: T;
  message?: string;
};

type QuizOption = {
  option_text: string;
  is_correct: boolean;
};

type QuizQuestion = {
  question_text: string;
  question_type: "mcq" | "multi_select" | "short_answer";
  order_index: number;
  options: QuizOption[];
};

function getErrorMessage(error: unknown, fallback: string): string {
  if (axios.isAxiosError(error)) {
    return error.response?.data?.message || fallback;
  }

  if (error instanceof Error) return error.message;

  return fallback;
}

const inputClass =
  "w-full rounded-lg border bg-background px-3 py-2 text-foreground";

const buttonClass =
  "rounded-lg border px-4 py-2 text-sm disabled:cursor-not-allowed disabled:opacity-50";

export default function InstructorQuizManagementPage() {
  const { id: courseId } = useParams<{ id: string }>();

  const [modules, setModules] = useState<Module[]>([]);
  const [quizzes, setQuizzes] = useState<Quiz[]>([]);
  const [selectedModuleId, setSelectedModuleId] = useState("");
  const [title, setTitle] = useState("");
  const [questionText, setQuestionText] = useState("");
  const [questionType, setQuestionType] =
    useState<QuizQuestion["question_type"]>("mcq");
  const [options, setOptions] = useState<QuizOption[]>([
    { option_text: "", is_correct: true },
    { option_text: "", is_correct: false },
  ]);

  const [loadingModules, setLoadingModules] = useState(true);
  const [loadingQuizzes, setLoadingQuizzes] = useState(true);
  const [submitting, setSubmitting] = useState(false);
  const [updatingQuizId, setUpdatingQuizId] = useState("");
  const [error, setError] = useState("");
  const [message, setMessage] = useState("");

  const loadModules = useCallback(async () => {
    if (!courseId) {
      setError("Course ID is missing.");
      setLoadingModules(false);
      return;
    }

    try {
      const response = await api.get<ApiResponse<{ modules?: Module[] }>>(
        `/courses/${courseId}/manage`,
      );

      if (!response.data?.success) {
        throw new Error("Failed to load course modules.");
      }

      const loadedModules = [...(response.data.data?.modules ?? [])].sort(
        (a, b) => a.order_index - b.order_index,
      );

      setModules(loadedModules);
      setSelectedModuleId((current) =>
        loadedModules.some((module) => module.id === current)
          ? current
          : (loadedModules[0]?.id ?? ""),
      );
    } catch (err) {
      setError(getErrorMessage(err, "Failed to load course modules."));
    } finally {
      setLoadingModules(false);
    }
  }, [courseId]);

  const loadQuizzes = useCallback(async () => {
    try {
      const response = await api.get<ApiResponse<Quiz[]>>(
        "/quizzes/instructor",
      );

      if (!response.data?.success) {
        throw new Error("Failed to load quizzes.");
      }

      setQuizzes(response.data.data ?? []);
    } catch (err) {
      setError(getErrorMessage(err, "Failed to load quizzes."));
    } finally {
      setLoadingQuizzes(false);
    }
  }, []);

  useEffect(() => {
    void loadModules();
    void loadQuizzes();
  }, [loadModules, loadQuizzes]);

  function clearNotifications() {
    setError("");
    setMessage("");
  }

  function handleQuestionTypeChange(nextType: QuizQuestion["question_type"]) {
    setQuestionType(nextType);

    if (nextType !== "short_answer" && options.length < 2) {
      setOptions([
        { option_text: "", is_correct: true },
        { option_text: "", is_correct: false },
      ]);
    }

    clearNotifications();
  }

  function updateOption(
    index: number,
    field: "option_text" | "is_correct",
    value: string | boolean,
  ) {
    setOptions((current) =>
      current.map((option, optionIndex) => {
        if (optionIndex !== index) return option;

        if (field === "option_text") {
          return { ...option, option_text: value as string };
        }

        return { ...option, is_correct: value as boolean };
      }),
    );
  }

  function addOption() {
    setOptions((current) => [
      ...current,
      { option_text: "", is_correct: false },
    ]);
  }

  function removeOption(index: number) {
    setOptions((current) =>
      current.filter((_, optionIndex) => optionIndex !== index),
    );
  }

  async function handleCreateQuiz(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    clearNotifications();

    if (submitting) return;

    if (!selectedModuleId) {
      setError("Select a module before creating a quiz.");
      return;
    }

    const cleanTitle = title.trim();
    const cleanQuestion = questionText.trim();

    if (!cleanTitle || !cleanQuestion) {
      setError("Enter both a quiz title and a question.");
      return;
    }

    let questionOptions: QuizOption[] = [];

    if (questionType !== "short_answer") {
      questionOptions = options.map((option) => ({
        option_text: option.option_text.trim(),
        is_correct: option.is_correct,
      }));

      if (
        questionOptions.length < 2 ||
        questionOptions.some((option) => !option.option_text) ||
        new Set(questionOptions.map((option) => option.option_text)).size !==
          questionOptions.length ||
        !questionOptions.some((option) => option.is_correct)
      ) {
        setError(
          "Add at least two unique, non-empty options and mark a correct answer.",
        );
        return;
      }

      if (
        questionType === "mcq" &&
        questionOptions.filter((option) => option.is_correct).length !== 1
      ) {
        setError(
          "Multiple-choice questions must have exactly one correct option.",
        );
        return;
      }
    }

    setSubmitting(true);

    try {
      const response = await api.post<ApiResponse<Quiz>>("/quizzes/", {
        module_id: selectedModuleId,
        title: cleanTitle,
        questions: [
          {
            question_text: cleanQuestion,
            question_type: questionType,
            order_index: 1,
            options: questionOptions,
          },
        ],
      });

      if (!response.data?.success) {
        throw new Error(response.data?.message || "Quiz creation failed.");
      }

      setTitle("");
      setQuestionText("");
      setQuestionType("mcq");
      setOptions([
        { option_text: "", is_correct: true },
        { option_text: "", is_correct: false },
      ]);
      setMessage("Quiz created successfully.");

      await loadQuizzes();
    } catch (err) {
      setError(getErrorMessage(err, "Failed to create quiz."));
    } finally {
      setSubmitting(false);
    }
  }

  async function handleApprovalChange(
    quizId: string,
    approvalStatus: "approved" | "rejected",
  ) {
    clearNotifications();
    setUpdatingQuizId(quizId);

    try {
      const response = await api.patch<ApiResponse<Quiz>>(
        `/quizzes/${quizId}/approval`,
        { approval_status: approvalStatus },
      );

      if (!response.data?.success) {
        throw new Error(response.data?.message || "Quiz status update failed.");
      }

      setQuizzes((current) =>
        current.map((quiz) =>
          quiz.id === quizId
            ? { ...quiz, approval_status: approvalStatus }
            : quiz,
        ),
      );

      setMessage(`Quiz ${approvalStatus} successfully.`);
    } catch (err) {
      setError(getErrorMessage(err, "Failed to update quiz status."));
    } finally {
      setUpdatingQuizId("");
    }
  }

  const courseQuizzes = quizzes.filter((quiz) =>
    modules.some((module) => module.id === quiz.module_id),
  );

  return (
    <main className="min-h-screen bg-background px-4 py-8 text-foreground sm:px-6">
      <div className="mx-auto max-w-4xl space-y-8">
        <Link
          to="/instructor/courses"
          className="text-sm text-muted-foreground hover:underline"
        >
          ← Back to My Courses
        </Link>

        <header>
          <h1 className="mt-4 text-3xl font-bold">Quiz Management</h1>
          <p className="mt-2 text-sm text-muted-foreground">
            Create quizzes for your course and manage their approval status.
          </p>
          {courseId && (
            <p className="mt-1 break-all text-xs text-muted-foreground">
              Course ID: {courseId}
            </p>
          )}
        </header>

        {error && (
          <p
            role="alert"
            className="rounded-lg border border-destructive p-3 text-sm"
          >
            {error}
          </p>
        )}

        {message && (
          <p role="status" className="rounded-lg border p-3 text-sm">
            {message}
          </p>
        )}

        <section className="space-y-5 rounded-xl border bg-card p-5 sm:p-6">
          <h2 className="text-xl font-semibold">Create a Manual Quiz</h2>

          {loadingModules ? (
            <p>Loading course modules...</p>
          ) : modules.length === 0 ? (
            <p className="text-sm text-muted-foreground">
              No modules found. Create a module in Course Content Manager first.
            </p>
          ) : (
            <form onSubmit={handleCreateQuiz} className="space-y-5">
              <div>
                <label htmlFor="quiz-module" className="mb-2 block font-medium">
                  Course module
                </label>
                <select
                  id="quiz-module"
                  required
                  value={selectedModuleId}
                  onChange={(event) => setSelectedModuleId(event.target.value)}
                  className={inputClass}
                >
                  {modules.map((module) => (
                    <option key={module.id} value={module.id}>
                      {module.order_index}. {module.title}
                    </option>
                  ))}
                </select>
              </div>

              <div>
                <label htmlFor="quiz-title" className="mb-2 block font-medium">
                  Quiz title
                </label>
                <input
                  id="quiz-title"
                  required
                  maxLength={200}
                  value={title}
                  onChange={(event) => setTitle(event.target.value)}
                  placeholder="Introduction Quiz"
                  className={inputClass}
                />
              </div>

              <div>
                <label
                  htmlFor="quiz-question"
                  className="mb-2 block font-medium"
                >
                  Question 1
                </label>
                <textarea
                  id="quiz-question"
                  required
                  maxLength={5000}
                  rows={3}
                  value={questionText}
                  onChange={(event) => setQuestionText(event.target.value)}
                  placeholder="Enter your question"
                  className={inputClass}
                />
              </div>

              <div>
                <label
                  htmlFor="quiz-question-type"
                  className="mb-2 block font-medium"
                >
                  Question type
                </label>
                <select
                  id="quiz-question-type"
                  value={questionType}
                  onChange={(event) =>
                    handleQuestionTypeChange(
                      event.target.value as QuizQuestion["question_type"],
                    )
                  }
                  className={inputClass}
                >
                  <option value="mcq">
                    Multiple choice (one correct answer)
                  </option>
                  <option value="multi_select">
                    Multi-select (multiple correct answers)
                  </option>
                  <option value="short_answer">Short answer</option>
                </select>
              </div>

              {questionType !== "short_answer" && (
                <div className="space-y-3">
                  <h3 className="font-medium">Answer options</h3>
                  <p className="text-sm text-muted-foreground">
                    {questionType === "mcq"
                      ? "Select exactly one correct answer."
                      : "Select one or more correct answers."}
                  </p>

                  {options.map((option, index) => (
                    <div
                      key={index}
                      className="space-y-2 rounded-lg border p-3"
                    >
                      <label
                        htmlFor={`quiz-option-${index}`}
                        className="text-sm font-medium"
                      >
                        Option {index + 1}
                      </label>

                      <input
                        id={`quiz-option-${index}`}
                        required
                        maxLength={1000}
                        value={option.option_text}
                        onChange={(event) =>
                          updateOption(index, "option_text", event.target.value)
                        }
                        placeholder={`Enter option ${index + 1}`}
                        className={inputClass}
                      />

                      <label className="flex items-center gap-2 text-sm">
                        <input
                          type={questionType === "mcq" ? "radio" : "checkbox"}
                          name="correct-option"
                          checked={option.is_correct}
                          onChange={() => {
                            if (questionType === "mcq") {
                              setOptions((current) =>
                                current.map((item, optionIndex) => ({
                                  ...item,
                                  is_correct: optionIndex === index,
                                })),
                              );
                            } else {
                              updateOption(
                                index,
                                "is_correct",
                                !option.is_correct,
                              );
                            }
                          }}
                        />
                        Correct answer
                      </label>

                      {options.length > 2 && (
                        <button
                          type="button"
                          onClick={() => removeOption(index)}
                          className="text-sm text-destructive hover:underline"
                        >
                          Remove option
                        </button>
                      )}
                    </div>
                  ))}

                  <button
                    type="button"
                    onClick={addOption}
                    className={buttonClass}
                  >
                    + Add option
                  </button>
                </div>
              )}

              <button
                type="submit"
                disabled={submitting || loadingModules || modules.length === 0}
                className="w-full rounded-lg bg-primary px-4 py-3 font-medium text-primary-foreground disabled:opacity-50"
              >
                {submitting ? "Creating quiz..." : "Create Quiz"}
              </button>
            </form>
          )}
        </section>

        <section className="space-y-4 rounded-xl border bg-card p-5 sm:p-6">
          <div>
            <h2 className="text-xl font-semibold">Existing Quizzes</h2>
            <p className="mt-1 text-sm text-muted-foreground">
              Showing quizzes belonging to this course.
            </p>
          </div>

          {loadingQuizzes || loadingModules ? (
            <p>Loading quizzes...</p>
          ) : courseQuizzes.length === 0 ? (
            <p className="text-sm text-muted-foreground">
              No quizzes found for this course yet.
            </p>
          ) : (
            <div className="space-y-3">
              {courseQuizzes.map((quiz) => {
                const module = modules.find(
                  (item) => item.id === quiz.module_id,
                );

                return (
                  <article
                    key={quiz.id}
                    className="space-y-3 rounded-lg border p-4"
                  >
                    <div className="flex flex-wrap items-start justify-between gap-3">
                      <div className="min-w-0">
                        <h3 className="break-words font-semibold">
                          {quiz.title}
                        </h3>
                        <p className="mt-1 text-sm text-muted-foreground">
                          Module: {module?.title ?? "Unknown module"}
                        </p>
                        <p className="text-sm text-muted-foreground">
                          Type:{" "}
                          {quiz.is_ai_generated ? "AI generated" : "Manual"}
                        </p>
                      </div>

                      <span className="rounded-full border px-3 py-1 text-xs capitalize">
                        {quiz.approval_status}
                      </span>
                    </div>

                    {quiz.approval_status === "pending" && (
                      <div className="flex flex-wrap gap-2">
                        <button
                          type="button"
                          disabled={updatingQuizId === quiz.id}
                          onClick={() =>
                            void handleApprovalChange(quiz.id, "approved")
                          }
                          className="rounded-lg bg-primary px-3 py-2 text-sm text-primary-foreground disabled:opacity-50"
                        >
                          {updatingQuizId === quiz.id
                            ? "Updating..."
                            : "Approve"}
                        </button>

                        <button
                          type="button"
                          disabled={updatingQuizId === quiz.id}
                          onClick={() =>
                            void handleApprovalChange(quiz.id, "rejected")
                          }
                          className="rounded-lg border border-destructive px-3 py-2 text-sm text-destructive disabled:opacity-50"
                        >
                          Reject
                        </button>
                      </div>
                    )}

                    {quiz.approval_status !== "pending" && (
                      <button
                        type="button"
                        disabled={updatingQuizId === quiz.id}
                        onClick={() =>
                          void handleApprovalChange(
                            quiz.id,
                            quiz.approval_status === "approved"
                              ? "rejected"
                              : "approved",
                          )
                        }
                        className={buttonClass}
                      >
                        Change to{" "}
                        {quiz.approval_status === "approved"
                          ? "Rejected"
                          : "Approved"}
                      </button>
                    )}
                  </article>
                );
              })}
            </div>
          )}
        </section>
      </div>
    </main>
  );
}

  async (error: AxiosError) => {
    const originalRequest = error.config as RetryableRequestConfig | undefined;

    if (
      error.response?.status !== 401 ||
      !originalRequest ||
      originalRequest._retry ||
      originalRequest.url?.includes("/auth/login") ||
      originalRequest.url?.includes("/auth/refresh") ||
      originalRequest.url?.includes("/auth/logout")
    ) {
      return Promise.reject(error);
    }

    originalRequest._retry = true;

    try {
      if (!refreshPromise) {
        refreshPromise = refreshClient
          .post<RefreshResponse>("/auth/refresh")
          .then((response) => {
            const result = response.data;
            const newAccessToken = result.data?.accessToken;

            if (!result.success || !newAccessToken) {
              throw new Error("Token refresh failed.");
            }

            setAccessToken(newAccessToken);
            return newAccessToken;
          })
          .finally(() => {
            refreshPromise = null;
          });
      }

      const newAccessToken = await refreshPromise;

      originalRequest.headers.set("Authorization", `Bearer ${newAccessToken}`);

      return api(originalRequest);
    } catch (refreshError) {
      clearTokens();
      return Promise.reject(refreshError);
    }
  },
);

export default api;

  baseURL: import.meta.env.VITE_API_URL,
  withCredentials: true,
  headers: {
    "Content-Type": "application/json",
  },
});

interface RefreshResponse {
  success: boolean;
  data?: {
    accessToken: string;
  };
}

interface MeResponse {
  success: boolean;
  data?: {
    id: string;
    full_name: string;
    email: string;
    roles: string[];
  };
}

export async function restoreSession(): Promise<void> {
  try {
    const refreshResponse =
      await refreshClient.post<RefreshResponse>("/auth/refresh");

    const accessToken = refreshResponse.data?.data?.accessToken;

    if (!refreshResponse.data.success || !accessToken) {
      throw new Error("Session refresh failed");
    }

    setAccessToken(accessToken);

    const meResponse = await api.get<MeResponse>("/auth/me");
    const user = meResponse.data?.data;

    if (!meResponse.data.success || !user) {
      throw new Error("Unable to restore user profile");
    }

    store.dispatch(setUser(user));
  } catch {
    clearTokens();
    store.dispatch(clearAuth());
  }
}

});
